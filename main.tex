\section{Part I: Fundamentals}
\subsection{Definitions}
\begin{definition}[Computer Vision]
Inverse algorithmic process to infer properties of the world based on sensory observations.
\end{definition}
\begin{definition}[Autonomous Car (ADS-DV)]
ADS-equipped vehicle designed for driverless operation within its given operational design domain.
\end{definition}
\begin{definition}[Automated Driving System (ADS)]
Hardware and software collectively capable of performing the entire DDT on a sustained basis, possibly limited to a specific ODD.
\end{definition}
\begin{definition}[Operational Design Domain (ODD)]
Conditions (environmental, geographical, temporal, traffic/roadway) under which a given ADS is designed to function.
\end{definition}
\begin{definition}[Dynamic Driving Task (DDT)]
All real-time operational and tactical functions for on-road driving (excludes strategic functions). Subtasks: (1) lateral control, (2) longitudinal control, (3) OEDR monitoring, (4) OEDR execution, (5) maneuver planning, (6) conspicuity enhancement.
\end{definition}
\begin{definition}[OEDR]
Object and Event Detection and Response. DDT subtasks 3--4: monitor driving environment via detection, recognition, classification, response preparation, and response execution.
\end{definition}
\begin{definition}[Driving Automation System]
Hardware and software capable of performing part or all of the DDT. $\text{ADS} \subseteq \text{Driving Automation Systems}$. Four core modules: (1) semantic and geometric \textbf{Perception} (2) \textbf{Prediction} from temporal perception (3) path and motion \textbf{Planning} via response determination (4) \textbf{Control}
\end{definition}

\subsection{Optical Autonomous Driving Sensors}
Countless sensor choices exist. We make the distinction of \textbf{external} (\textit{camera, lidar, radar, event camera, GPS, IMU, speedometer, ...}) and \textbf{internal} (\textit{camera, thermal camera, microphone}) sensors. Our focus further lies on the \textit{Optical Sensors}.\\

Optical Sensors can be categorized along the following features: 
\begin{itemize}
\item \textit{Wavelength}: spectral band of light the sensor operates in (e.g.\ visible, near-infrared, millimeter-wave).
\item \textit{Active or Passive}: whether the sensor emits its own signal (active) or relies on external illumination (passive).
\item \textit{Resolution}: spatial density of measurements; determines fineness of detail captured.
\item \textit{Latency / Frame Rate}: temporal frequency at which full measurements are acquired.
\item \textit{Field of View} (FOV): angular extent of the observable scene, horizontal and vertical.
\item \textit{Range}: maximum distance at which the sensor can reliably detect objects.
\end{itemize}

\subsubsection{Light Detection and Ranging (Lidar)}
\textbf{Characteristics}: Active sensor, near-infrared wavelength (905nm), direct range measurements via time of flight (TOF), i.e., timing of light emission until recapture of photons, large horizontal and fair vertical FOV (3D), fair resolution

\paragraph{Pulsed Lidar}
Multiple short laser pulses emitted in different directions and at different times. Each measurement: pulse emitter $\to$ beam steering $\to$ target $\to$ photodetector (APD). A timer measures the round-trip delay to infer range $R_0$

\end{figure}

\subparagraph{Beam Configurations.}
Real beams diverge; receivers have finite FOV. Two configurations:
\begin{itemize}
    \item \textit{Coaxial}: transmitter and receiver share a common optical axis.
    \item \textit{Bistatic}: optical axes of transmitter and receiver are parallel but offset.
     The \textbf{crossover function} $\xi(R) \in [0,1]$ gives the fraction of the illuminated area observed by the receiver as a function of range. For bistatic: $\xi(R)=0$ at short range (no crossover), $\xi(R)=1$ beyond some $R_2$ (full crossover).
\end{itemize}

    \caption{Lidar Bistatic Beam Configuration}
    \label{fig:LidarBistatic}
\end{figure}

\subparagraph{Pulse Waveform (Sinusoidal Form).}
\begin{equation*}
    P_T(t) = \begin{cases} P_0 \sin^2\!\left(\dfrac{\pi}{2\tau_H}\, t\right), & 0 \leq t \leq 2\tau_H, \\ 0, & \text{otherwise.} \end{cases}
\end{equation*}
where $P_T(t)$: emitted radiant power, $P_0$: peak power, $\tau_H$: half-power pulse width ($10$--$20$\,ns), $c\tau_H$: spatial half-power width ($3$--$6$\,m).\\
Physically more realistic is the square-exponential form (left heavy, right tailed) -- omitted here. 

\subparagraph{Linear System Model.}
Received radiant power as a convolution of emitted power with the optical system impulse response:
\begin{equation*}
    P_R(R) = C_A \int_0^{2R/c} P_T(t)\, H\!\left(R - \frac{ct}{2}\right) dt
\end{equation*}
where $P_R(R)$: range-dependent received power, $H(R)$: impulse response of the optical system, $C_A$: system constant (independent of time and range). $ct/2$ maps time $t$ to the corresponding range (round-trip).
 
\textbf{Intuition:} $H(R)$ is what we would receive from a perfectly concentrated (Dirac) pulse; the actual response is a smeared version due to finite pulse width - ambiguity from temporal profile/ early or late photon.

\subparagraph{Impulse Response Decomposition.}
\begin{equation*}
    H(R) = H_C(R)\, H_T(R)
\end{equation*}
\begin{itemize}
    \item \textit{Optical channel} (medium): $H_C(R) = \dfrac{T^2(R)}{R^2}\,\xi(R)$, \quad where $T(R) = \exp(-\alpha R)$ is the one-way transmission loss ($\alpha$ depends on visibility, assumes homogeneous scattering medium). Similar to what we see in NeRF or Fog Simulation, but squared, since light must travel back and forth.
    \item \textit{Target}: $H_T(R) = \rho_0\,\delta(R - R_0)$ in clear weather, where $\rho_0$ is the differential reflectivity.
\end{itemize}
 
\subparagraph{Clear Weather Simplification.}
With $\alpha = 0$ (no attenuation -- clear air) and $\xi(R_0)=1$:
\begin{equation*}
    H(R) = \frac{1}{R_0^2}\,\rho_0\,\delta(R - R_0)
\end{equation*}
Substituting into the linear system model yields:
\begin{equation*}
    P_R(R) = \begin{cases} \dfrac{C_A P_0 \rho_0}{R_0^2}\, \sin^2\!\left(\dfrac{\pi(R-R_0)}{c\tau_H}\right), & R_0 \leq R \leq R_0 + c\tau_H, \\ 0, & \text{otherwise.} \end{cases}
\end{equation*}
 
\textbf{Intuition:} The target acts as a Dirac delta in range, so the convolution merely shifts and scales the emitted waveform — the received pulse preserves the shape of $P_T$, centered at $R_0$. The $1/R_0^2$ decay is the inverse-square law: emitted power spreads over a sphere of area $\propto R_0^2$. The proportionality to $\rho_0$ is why lidar returns per-point \textit{intensity} alongside range. The pulse occupies a range interval of width $c\tau_H$, so shorter pulses yield finer range resolution at the cost of less energy per pulse.

\subparagraph{Features of Pulsed Lidar.}
\begin{itemize}
    \item \textit{Ranging precision}: limited by temporal resolution of timing electronics. Typical $\sim 0.1$\,ns $\Rightarrow$ $\sim 1.5$\,cm.
    \item \textit{Maximum range}: limited by signal to noise ratio (SNR) (attenuation + detector noise/jitter): $100$--$200$\,m.
    \item \textit{Range--rate trade-off}: maximum unambiguous range is inversely proportional to pulse repetition rate (aliasing) -- upper limit of the integral
\end{itemize}

\subparagraph{Acquisition of 3D Point Clouds.}
Each pulse yields a single range measurement along a direction $(\alpha, \epsilon)$. To obtain a 3D point cloud, the sensor scans across a range of azimuth $\alpha$ and elevation $\epsilon$ values, which jointly define the FOV. Spherical-to-Cartesian conversion:
\begin{equation*}
    \begin{bmatrix} x \\ y \\ z \end{bmatrix}
    = \mathbf{h}^{-1}(r, \alpha, \epsilon)
    = \begin{bmatrix} r \cos\alpha \cos\epsilon \\ r \sin\alpha \cos\epsilon \\ r \sin\epsilon \end{bmatrix}
\end{equation*}
Each point additionally carries an \textit{intensity} value $= f(P_R(R_0))$, derived from the peak received power.
 
\subparagraph{Scanning Mechanisms.}
Taxonomy: non-scanning (flash) vs.\ scanning lidar. Scanning subdivides into non-mechanical (OPA, solid-state) and mechanical (MEMS, motorized optomechanical). Driving-relevant types are motorized mechanical and MEMS scanners.\\
 
\textit{Motorized mechanical}: nodding-mirror tilt controls vertical FOV; rotating housing ($>1000$ RPM) provides panoramic $360^\circ$ horizontal FOV. Modern systems use laser/photodiode arrays $\Rightarrow$ mechanical rotation only along azimuth. Horizontal angular resolution up to $0.1^\circ$, max range up to $200$\,m. \\
 
\textit{MEMS}: tiny oscillating mirrors (few mm diameter) steer individual pulses. 1D MEMS scans one angular dimension (needs laser array or motor for the other); 2D MEMS scans both. Resonant scanning: frequency $=$ resonant frequency of mirror, high frame rate, sensitive to environmental variation. Non-resonant: more robust, lower frame rate.
 
\begin{center}
\begin{tabular}{l|l}
    \textbf{Motorized Mechanical} & \textbf{MEMS} \\ \hline
    Panoramic $360^\circ$ horizontal FOV & Limited ($\sim 120^\circ$) horizontal FOV \\
    Uniform azimuthal scanning pattern & Complex / less interpretable pattern \\
    Bulky, vulnerable to wear & Small, no mechanical wear \\
    Imbalanced vert.\ vs.\ horiz.\ resolution & Balanced resolution \\
    Low frame rate ($10$--$20$\,Hz) & Slightly higher frame rate \\
    Expensive & Less expensive \\
\end{tabular}
\end{center}
 
\textbf{Measurement Errors in Lidar.}
\begin{itemize}
    \item \textit{Sensor}: timing jitter in arrival detection; rotary encoder uncertainty (angular error $\to$ distortion).
    \item \textit{Scene}: range ambiguity (multiple waveform maxima); non-Lambertian surfaces (absorptive $\to$ no return, specular $\to$ multi-path); motion distortion from finite frame rate (not rectifiable, unlike ego-motion).
    \item \textit{Optical medium}: attenuation (e.g.\ fog $\to$ reduced peak amplitude); backscattering (e.g.\ snow/rain $\to$ false peaks, spurious points).
    \item \textit{Light sources}: ambient NIR interference (minor); cross-talk from other lidars.
\end{itemize}

\subsubsection{Camera}
\textbf{Characteristics}: Passive sensor, visible spectrum (typically three sensitivity profiles / channels), very high resolution, fair frame rate, moderate FOV for perspective cameras, cheap.

\paragraph{Pinhole Camera Model.}
Ideal pinhole $=$ center of projection $\mathbf{O}_c$. All scene rays pass through a single point onto the image plane. Virtual image plane (in front of pinhole): image is not inverted. This yields a \textit{perspective projection}.

\paragraph{Homogeneous Coordinates.}
Unify representation of points and planes ($ax+by+cz-d=0$) by adding one dimension:
\begin{equation*}
    \mathbf{P} = \begin{pmatrix} x \\ y \\ z \\ 1 \end{pmatrix} \quad \text{(point)}, \qquad
    \boldsymbol{\Pi} = \begin{pmatrix} a \\ b \\ c \\ -d \end{pmatrix} \quad \text{(plane)},
\end{equation*}
where $\boldsymbol{\Pi} \cdot \mathbf{P} = 0$ iff $\mathbf{P}$ lies on $\boldsymbol{\Pi}$.


\paragraph{Geometric Transformations.}
Notation: ${}^{F}\mathbf{P}$ denotes coordinates of point $\mathbf{P}$ in frame $F$.
\begin{itemize}
    \item \textit{Translation}: ${}^{B}\mathbf{P} = {}^{A}\mathbf{P} + {}^{B}\mathbf{O}_A$, where ${}^{B}\mathbf{O}_A$ is the origin of frame $A$ expressed in frame $B$.
    \item \textit{Rotation}: ${}^{B}\mathbf{P} = {}^{B}_{A}\mathbf{R}\;{}^{A}\mathbf{P}$, where ${}^{B}_{A}\mathbf{R} = ({}^{B}\mathbf{i}_A \;\; {}^{B}\mathbf{j}_A \;\; {}^{B}\mathbf{k}_A)$ and $\mathbf{R}^T\mathbf{R} = \mathbf{I}$.
    \item \textit{Rigid / Euclidean}: ${}^{B}\mathbf{P} = {}^{B}_{A}\mathbf{R}\;{}^{A}\mathbf{P} + {}^{B}\mathbf{O}_A$. In homogeneous coordinates:
\end{itemize}
\begin{equation*}
    \begin{bmatrix} {}^{B}\mathbf{P} \\ 1 \end{bmatrix}
    = \begin{bmatrix} {}^{B}_{A}\mathbf{R} & {}^{B}\mathbf{O}_A \\ \mathbf{0}^T & 1 \end{bmatrix}
    \begin{bmatrix} {}^{A}\mathbf{P} \\ 1 \end{bmatrix}
\end{equation*}

\paragraph{3D to 2D Perspective Projection.}
Given a 3D point $\mathbf{P}_c = (x_c, y_c, z_c)^T$ in camera coordinates, projection to 2D homogeneous pixel coordinates $\tilde{\mathbf{x}}_s$:
\begin{equation*}
    \tilde{\mathbf{x}}_s \sim \mathbf{K}\, \mathbf{P}_c, \qquad
    \mathbf{K} = \begin{bmatrix} f_x & s & c_x \\ 0 & f_y & c_y \\ 0 & 0 & 1 \end{bmatrix}
\end{equation*}
where $f_x, f_y$: focal lengths, $s$: skew, $(c_x, c_y)$: principal point, $K$ encoding the camera intrinsics (5 DOF)\\
 
\textbf{Intuition:} Derived from similar triangles starting with: $-(x_s - c_x)/x_c = f/(-z_c)$.\\
 
\textbf{Incorporating Extrinsics.} rigid transformation from world frame to camera frame: $ \mathbf{P}_c = [\mathbf{R} \mid \mathbf{t}]\, \tilde{\mathbf{P}}_w$

\paragraph{Complete Projection.} from 3D homogeneous world coordinates to 2D homogeneous image coordinates:
\begin{equation*}
    \tilde{\mathbf{x}}_s \sim \underbrace{\mathbf{K}\, [\mathbf{R} \mid \mathbf{t}]}_{\mathbf{C}}\; \tilde{\mathbf{P}}_w \qquad \text{ $\mathbf{C}$: $3 \times 4$ \textbf{camera matrix} (11 DOF: 5 intrinsics $+$ 6 extrinsics)
}
\end{equation*}

\paragraph{Photometric Image Formation.}
Digital imaging pipeline (scene radiance $\to$ optics $\to$ sensor $\to$ ISP $\to$ output), BRDF-based (Bidirectiional Reflectance Distribution Function) rendering equation, Lambertian vs.\ specular reflection. (details omitted)

\subsubsection{Radar}
\textbf{Characteristics}: Active sensor, $70$--$80$\,GHz (wavelengths $\sim$ few mm), measures \textit{range and velocity}. Robust to fog/rain/snow (wavelength $\gg$ particle size $\Rightarrow$ no scattering). Large horizontal FOV, very narrow vertical FOV. Low azimuthal resolution (few azimuth bins across the $360^\circ$ sweep), high range resolution. Low frame rate. Range: $200$--$600$\,m.

\paragraph{FMCW Scanning Radar.}
Two waveform types: fixed-frequency pulses and frequency-modulated continuous wave (FMCW). Output is discretized into $M$ azimuths $\times$ $N$ range bins; per-bin measurement is received power.\\
 
\textbf{Radar equation}: $P_{\text{received}} = \frac{P_{\text{transmitted}}\, G\, \sigma\, A_e}{16\pi^2\, r^4\, L}$
where $G$: transmitter gain, $\sigma$: radar cross section (RCS, depends on target area/orientation/material), $A_e$: effective antenna aperture, $L$: loss factor, $r$: range. Note $1/r^4$ decay (vs.\ $1/R_0^2$ for lidar) because radar uses separate opening angles for transmitter and receiver. This is the radar analogue of the pulsed lidar received-power model: it determines how SNR degrades with range and target properties, and thus the sensor's effective detection limits.\\

\textbf{FMCW Range and Velocity Measurement}:\\
\textit{Modulation pattern}: frequency of transmitted signal as a function of time.\\
\textit{Intermediate frequency (IF)}: difference between transmitted and received signal frequencies.\\
 
For a \textit{static target}, a single linear frequency sweep (chirp, i.e.\ frequency ramping at constant rate $df/dt$) yields range directly from the IF: $r = \frac{c\,\Delta f}{2\,(df/dt)}$ where $\Delta f$ is the IF.\\
 
For a \textit{moving target}, there is ambiguity between TOF shift and Doppler shift $\Delta f_v$. Resolved using repeating triangular modulation pattern (up-chirp IF $= \Delta f_1$, down-chirp IF $= \Delta f_2$):
\begin{equation}
    \Delta f_v = \frac{\Delta f_2 - \Delta f_1}{2}, \qquad
    \Delta f_r = \frac{\Delta f_1 + \Delta f_2}{2}
\end{equation}
$\Delta f_r$ gives range; $\Delta f_v$ gives radial velocity relative to sensor.\\

\textbf{Range-FFT}: FFT of the IF signal yields the power-range spectrum for each azimuth. Each frequency component in the IF corresponds to a specific range, so the FFT separates targets at different distances. Real targets are not point-like $\Rightarrow$ energy spreads across multiple range bins, unlike the ideal single-peak response.

\subsubsection{Sensor Distribution on Car}
Typical ADS platform uses $20$--$40$ sensors. Sensor placement affects covered FOV, cost, reliability, dirt management, serviceability, and crash survivability. Roof-mounted setups (common for robotaxis / data collection) are not acceptable for commercial vehicles. Changes in sensor 6D pose introduce dataset bias, potentially requiring full re-collection. Sensor suite composition is still under active definition in the industry.

\section{Part II: Semantic Perception}
\subsection{Semantic Segmentation}
\subsubsection{Definition and Basics}
\begin{definition}[Semantic Image Segmentation]
Given a predefined class set $\mathcal{C} = \{1, \dots, C\}$, assign every pixel $\mathbf{x}_i$ of an input image $\mathbf{I} \in \mathbb{R}^{H \times W \times 3}$ a single class label $y_i \in \mathcal{C}$. No differentiation between distinct instances of the same class (cf.\ instance/panoptic segmentation).
\end{definition}

\paragraph{Data Format.}
Supervised setup: dataset $\mathcal{D} = \{(\mathbf{I}^{(n)}, \mathbf{Y}^{(n)})\}_{n=1}^N$.
\begin{itemize}
    \item \textit{Input}: RGB image $\mathbf{I} \in \mathbb{R}^{H \times W \times 3}$.
    \item \textit{Output / ground truth}: one-hot tensor $\mathbf{Y} \in \{0,1\}^{H \times W \times C}$ with $\sum_{c=1}^{C} Y_{h,w,c} = 1$ for every pixel $(h,w)$. One channel per class.
\end{itemize}
Indexing convention: $h \in \{1,\dots,H\}$ row, $w \in \{1,\dots,W\}$ column, $c \in \{1,\dots,C\}$ class channel, $k$ used as a dummy class index in sums.

\paragraph{Prediction Format.}
Model outputs a logits tensor $\mathbf{Z} \in \mathbb{R}^{H \times W \times C}$. Per-pixel softmax gives the soft prediction tensor $\hat{\mathbf{Y}} \in [0,1]^{H \times W \times C}$:
\begin{equation*}
    \hat{Y}_{h,w,c} = \frac{\exp(Z_{h,w,c})}{\sum_{k=1}^{C} \exp(Z_{h,w,k})}, \qquad \hat{\mathbf{Y}}_{h,w,:} \in \Delta^{C-1}, \qquad \text{Hard prediction: $\hat{y}_{h,w} = \arg\max_c \hat{Y}_{h,w,c}$.}
\end{equation*}

\paragraph{Categorical Cross-Entropy Loss.}
\begin{equation*}
    \mathcal{L}(\mathbf{I}) = -\frac{1}{HW} \sum_{h=1}^{H} \sum_{w=1}^{W} \sum_{c=1}^{C} Y_{h,w,c}\, \log \hat{Y}_{h,w,c}.
\end{equation*}
\textbf{Intuition:} One-hot $\mathbf{Y}$ collapses inner $c$-sum to the single true-class term ($\log \hat{Y}_{h,w,\, y_{h,w}}$) $\rightarrow$ average over $HW$ pixels.

\paragraph{Evaluation Metrics.}
\begin{itemize}
    \item \textit{Pixel accuracy}: fraction of correctly classified pixels. Skewed by frequent classes (sky, road).
    \item \textit{IoU (class $c$)}: $\mathrm{IoU}_c = \dfrac{|\hat{P}_c \cap P_c|}{|\hat{P}_c \cup P_c|}$, with $\hat{P}_c, P_c$ the predicted/GT pixel sets.
    \item \textit{mIoU}: $\frac{1}{C}\sum_c \mathrm{IoU}_c$. More stringent than pixel accuracy; mean over classes prevents large object bias (a wrong prediction will be wrong in all instances).
\end{itemize}

\subsubsection{Convolutional Neural Networks Primer}
\paragraph{Limitations of MLPs on Images.}
Vectorizing an $H \times W$ image as MLP input
\begin{enumerate}
    \item destroys 2D structure and local correlations; spatial invariances must be relearned;
    \item explodes the parameter count (e.g.\ $400\!\times\!250$ image, 10k hidden $\Rightarrow$ $\sim\!10^9$ parameters in the first layer alone).
\end{enumerate}

\paragraph{Discrete 2D Convolution.}
Slide the kernel $\mathbf{K} \in \mathbb{R}^{m \times n}$ over input $\mathbf{X}$; at each location multiply the overlapping pixel--weight pairs and sum:
\begin{equation*}
    (\mathbf{X} * \mathbf{K})[p, q] = \sum_{i=1}^{m} \sum_{j=1}^{n} X[p+i-1,\, q+j-1]\, K[i, j].
\end{equation*}
Formally a \textit{cross-correlation}; it differs from convolution proper by a kernel flip.

\textbf{Intuition:} The operation is \textit{local}, \textit{linear}, and \textit{shift-equivariant} (commutes with translation). The same parameters are used across all spatial locations -- \textit{parameter sharing}. The 2D structure of the input is preserved in the output.

\paragraph{Convolutional Layer: General Formulation.}
At layer $\ell$, the pre-activation tensor of the previous layer
\begin{equation*}
    \mathbf{z}_{\ell-1} \in \mathbb{R}^{H_{\ell-1} \times W_{\ell-1} \times d_{\ell-1}}
\end{equation*}
is convolved (after element-wise activation $g$) with $d_\ell$ \textit{3D kernels}
\begin{equation*}
    \mathbf{W}^{(\ell, c)} \in \mathbb{R}^{m_\ell \times n_\ell \times d_{\ell-1}}, \qquad c = 1, \dots, d_\ell,
\end{equation*}
each paired with a scalar bias $b_c^{(\ell)}$. One distinct kernel channel exists for every input channel; the kernel is therefore a 3D tensor whose depth matches $d_{\ell-1}$.

\subparagraph{Single Convolutional Neuron.}
For output channel $c$ at spatial location $(p, q)$, the pre-activation of layer $\ell$ is the sum of 2D cross-correlations against the activated outputs of the previous layer, plus the bias:
\begin{equation*}
    z_{\ell, c}[p, q] = \sum_{k=1}^{d_{\ell-1}} \sum_{i=1}^{m_\ell} \sum_{j=1}^{n_\ell} w^{(\ell)}_{k, c}[i, j]\; g\!\left( z_{\ell-1, k}[p+i-1,\, q+j-1] \right) + b_c^{(\ell)},
\end{equation*}
where $g(\cdot)$ is the (non-linear) activation function and $d_{\ell-1}$ denotes the number of input channels (i.e.\ the output channels of layer $\ell-1$, a hyperparameter of the preceding layer); $m_\ell, n_\ell$ are the kernel's spatial dimensions.\\
\textbf{Intuition:} The 2D cross-correlation generalizes to multi-channel inputs as a triple sum (two spatial $+$ one channel dim); the 3D kernel ``stacks'' the channel-wise correlations and the bias offsets the result. Parameters are shared across all spatial neurons of the same output channel.

\subparagraph{Element-wise Nonlinearity.}
The activation $g$ is applied independently to every scalar entry of the pre-activation tensor $\mathbf{z}_\ell$; the activated value
\begin{equation*}
    g\!\left( z_{\ell, c}[p, q] \right)
\end{equation*}
is what enters the next layer's convolution. The conv-and-bias step and the nonlinearity are conceptually separate sub-layers; $g$ couples neither spatial positions nor channels.

\subparagraph{Output Tensor.}
Stacking the $d_\ell$ feature maps gives the 3D pre-activation tensor
\begin{equation*}
    \mathbf{z}_\ell \in \mathbb{R}^{H_\ell \times W_\ell \times d_\ell},
\end{equation*}
i.e.\ multiple features per spatial location, one per output channel. $d_\ell$ is a layer hyperparameter

\subparagraph{Parameter Count.}
Per output channel: $m_\ell\, n_\ell\, d_{\ell-1}$ kernel weights $+$ $1$ bias. Total parameters in layer $\ell$:
\begin{equation*}
    \#\theta^{(\ell)} = d_\ell \,(\, m_\ell\, n_\ell\, d_{\ell-1} + 1\,).
\end{equation*}
\textbf{Intuition:} Cost scales with kernel footprint and the product of input/output channels, \textit{not} with the spatial size of the input -- this is what makes convolutional layers tractable on high-resolution images.

\paragraph{Pooling Layer.}
Non-parametric, (sub)differentiable downsampling. Hyperparameters: window size, stride. Max pooling provides robustness to local input variations and enables higher feature dimensionality without spatial-memory explosion.

\paragraph{Practical Variations.}
\begin{itemize}
    \item \textit{Padding}: valid convolution shrinks spatial dims. \textit{Same} mode pads by $\lfloor m/2 \rfloor$ (odd $m$) to preserve them. Modes: zero, replicate, reflect.
    \item \textit{Strided convolution}: kernel slides with step $s>1$. Motivation: faster receptive-field growth, spatial downsampling integrated into the conv. Often in early layers but pooling more common for downsampling. Higher stride $\rightarrow$ faster increase of receptive field size
\end{itemize}
Output spatial size (per dim):
\begin{equation*}
    H_{\mathrm{out}} = \left\lfloor \frac{H_{\mathrm{in}} + 2P - m}{s} \right\rfloor + 1,
\end{equation*}
where $H_{\mathrm{in}}$: input size, $m$: kernel size (per dim), $P$: padding width per border, $s$: stride.

\paragraph{Receptive Field.}
Region of the input on which a neuron's output depends. Grows with depth, kernel size, and stride. Deep CNN layers eventually attain global receptive fields covering the entire image.

\paragraph{CNN for Image Classification.}
Architecture: (conv $+$ activation $+$ pool)$^L \to$ flatten $\to$ FC layers $\to$ softmax. Final FC layers learn global interdependencies needed for the image-level prediction.

\subsubsection{Encoder--Decoder Paradigm}
\paragraph{Naive: Sliding-Window Classification.}
Extract a patch around each pixel, classify its center with a CNN. \\
\textit{Problems}: massive redundant computation; no global context (esp.\ for large objects). Does not scale to HR images.

\paragraph{End-to-End Architectures.}
Single forward pass from $\mathbf{I}$ to full label map $\hat{\mathbf{Y}}$. Goal: share computation across all spatial locations. Strategy: adapt classification CNNs to dense prediction.

\paragraph{Three Challenges.} (I) \textit{Reduced feature resolution} from repeated striding/pooling, (II) \textit{Multi-scale objects}, (III) \textit{Reduced localization accuracy} from CNN invariance

\paragraph{Fully Convolutional Networks (FCN).}
Replace the FC layers of a classification network with convolutional layers $\Rightarrow$ coarse high-level feature maps at low resolution. Per-pixel prediction at original image resolution then requires upsampling.

\paragraph{Naive Resolution Recovery.}
Remove pooling/striding entirely. Requires larger kernels for the same receptive field; every feature map $\mathbf{z}_\ell$ stays at $HW$ instead of shrinking geometrically, so activation memory $\sum_\ell H_\ell W_\ell d_\ell = HW \sum_\ell d_\ell$ scales linearly with depth (no geometric decay), and the same factor inflates per-layer FLOPs $\Rightarrow$ heavy compute/memory cost and overfitting risk.

\paragraph{Encoder--Decoder.}
\begin{itemize}
    \item \textit{Encoder}: maps $\mathbf{I}$ to a low-resolution, high-dimensional feature tensor; aggregates context.
    \item \textit{Decoder}: upsamples encoder features back to image resolution while reducing feature dimensionality.
    \item Spatial resolution contracts inside the encoder (high$\to$low) and is recovered by the decoder (low$\to$high); feature dimensionality follows the opposite trajectory.
\end{itemize}

\paragraph{In-Network Upsampling.}
\subparagraph{Unpooling.} During encoder max-pooling, store argmax positions in \textit{switch variables}; in the decoder, place each activation back at its recorded location. Produces an enlarged but \textit{sparse} feature map. Not learnable + symetric architecture for corresponding pairs.

\subparagraph{Transpose Convolution (Deconvolution).} Learnable upsampling for dense feature maps. Each input scalar weights a copy of the kernel; the weighted kernel is placed into the output at a location determined by the input position and the stride. Where output positions overlap across different input pixels, the contributions are summed. (Crop might be required to get exact target size)

\end{figure}

\subsubsection{Fundamental Architectures and Modules}
\paragraph{U-Net.}
Symmetric encoder--decoder with \textit{skip connections}: at each decoder stage, high-resolution encoder features are concatenated with upsampled decoder features along the channel dim, addressing the resolution challenge with fine grained localization information, followed by convolutions.
\begin{itemize}
    \item Contracting path: $\downarrow$ ``Where'', $\uparrow$ ``What'' (context).
    \item Expansive path: transpose convolutions $+$ concatenations recover spatial detail.
\end{itemize}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.95\linewidth]{img/unet.png}
    \label{fig:unet}
\end{figure}

\paragraph{Atrous (Dilated) Convolution.}
The \textit{effective} receptive field is smaller than the formal one: border connections are sparse. Atrous convolution inserts holes between kernel taps. For kernel $\mathbf{K} \in \mathbb{R}^{m \times n}$ and rate $r$:
\begin{equation*}
    (\mathbf{X} *_r \mathbf{K})[p, q] = \sum_{i=1}^{m} \sum_{j=1}^{n} X[p+ri,\, q+rj]\, K[i, j].
\end{equation*}
Standard convolution corresponds to $r=1$. Stacking atrous convolutions with growing $r$ yields \textit{exponential} expansion of receptive field size without resolution loss or coverage gaps and fixed number of weights. We use 0-inserts to retain tensor calculations.\\ 

\begin{figure}[H]
    \centering
    \includegraphics[width=0.85\linewidth]{img/atrous_rates.png}
    \caption{Atrous convolution (red dots: kernel taps)}
    \label{fig:atrous_rates}
\end{figure}

\subparagraph{Stride vs Atrous.}  Stride downsamples the output to grow receptive field; atrous spreads the kernel to grow receptive field without downsampling. That's why atrous is preferred for dense
  prediction (segmentation) — you keep spatial detail.


\paragraph{Attention.}
Provides a \textit{variable visual receptive field}. For tokens $\mathbf{X} \in \mathbb{R}^{N \times d}$, learned projections $\mathbf{W}_Q, \mathbf{W}_K, \mathbf{W}_V$ produce
\begin{equation*}
    \mathbf{Q} = \mathbf{X}\mathbf{W}_Q, \quad \mathbf{K} = \mathbf{X}\mathbf{W}_K, \quad \mathbf{V} = \mathbf{X}\mathbf{W}_V.
\end{equation*}
\begin{itemize}
    \item \textit{Query}: representation of the token used to compare it with other tokens.
    \item \textit{Key}: reference representation by which other tokens are compared against the token.
    \item \textit{Value}: output representation contributed by the token when it gets ``indexed''.
\end{itemize}
Scaled dot-product attention:
\begin{equation*}
    \mathrm{Attn}(\mathbf{Q}, \mathbf{K}, \mathbf{V}) = \mathrm{softmax}\!\left(\frac{\mathbf{Q}\mathbf{K}^\top}{\sqrt{d_k}}\right) \mathbf{V}.
\end{equation*}
Softmax taken along rows (per-query distribution). Division by $\sqrt{d_k}$ keeps gradient magnitudes well-scaled. Complexity: $\mathcal{O}(N^2 d)$ -- quadratic in tokens.\\

\textbf{Intuition:} Query/key match measures relevance of every other token; the row-softmax weights aggregate values $\Rightarrow$ each token is a learned linear mixture of all others. Globally connected receptive field at every layer.

\paragraph{Transformer Block.} Multi-head self-attention (concatenate $h$ parallel attention heads $\to$ linear projection) followed by a position-wise MLP. Residual connections and layer normalization around each sub-layer.

\paragraph{Transformers for Dense Prediction (Computer Vision).}
ViT-style large patches lose spatial precision $\Rightarrow$ poor for segmentation. Naive fix: restrict attention to local windows of $M \times M$ patches. Complexity becomes \textit{linear} in the number of patches (for fixed $M$). Drawback: receptive field bounded by the window across all layers.

\paragraph{Swin Transformer.}
\textit{Shifted-window} attention: alternate between W-MSA (window-based multi-head self-attention) and SW-MSA (shifted-window MSA). The window grid in SW-MSA is offset relative to W-MSA, so a query token's receptive field at layer $\ell+1$ is the union of layer-$\ell$ windows that overlap with the token's current local window.\\

\textbf{Intuition:} With enough shifts, local windowed attention reaches large receptive fields, keeping per-layer cost linear.

\subparagraph{Hierarchical Structure.}
\textit{Patch merging} concatenates features of each $2\!\times\!2$ neighboring patch group and applies a linear projection: tokens reduce by $4\times$ ($2\times$ spatial downsampling), output dim set to $2d$. Yields a multi-scale feature pyramid; Swin can serve as a drop-in CNN-backbone replacement for the encoder of standard segmentation networks.

\subsection{Object Detection}
\subsubsection{Fundamentals and Definitions}
\begin{definition}[Object Detection.] 
    Complement to semantic segmentation focussed on instance specific detection using bounding boxes. In essence the question of \textit{what?, where?}
\end{definition}

\paragraph{Evaluation.}
\begin{itemize}
    \item \textbf{Precision:} accuracy of predictions (how many correct?): $\frac{TP}{TP + FP}$
    \item \textbf{Recall:} detection of true objects: $\frac{TP}{TP + FN} $ 
    \item \textbf{IoU:} overlap of two bounding boxes (i.e., predicted vs ground truth)
    \item \textbf{(m)AP:} The \textbf{Average Precision} is the \textbf{Area Under the Curve} (Precision-Recall).
    Core idea is to take ranked detection by confidence level and some IoU threshold (when does a prediction count as correct).
    Then the \textbf{Precision-Recall Curve} is given by these ranked predictions. Since Prec and Rec $\in [0,1]$ the unit integral of AP is $\in [0,1]$.
    We ensure monotonic decrease by replacing each precision value with the maximum precision value to the right of that recall level.
    \item \textbf{Interpolated AP:} use an $n$-point interpolated AP, and then use the mean-AP of these $n$ points.
\end{itemize} 

\textit{AP Note:} Generally the AP is determined for a given class and iou-threshold.
Generally, we apply a number of predefined thresholds (loose to tight) and average them for a given class. 
All per class APs are average at the end for the full AP $\rightarrow$ \textit{average (class), average (IoU thresholds), average (precision @ different recall levels) precision}.

\subsubsection{Two Stage R-CNN}
\textit{Region-based} CNNs use per-region computation for the detection task, rather than considering the full input in one loss function target.
So the classifier only ever sees some number of potentially high quality regions, not the full $H_f W_f$ anchor grid.\\
\textit{Note:} One-Stage detectors, have no seperate proposal step. Every anchor is directly classified into $K+1$ classes and regresseed in a single forward pass (no proposal, every anchor, 1 box regression, faster, but lower quality).

\paragraph{R-CNN}
\begin{enumerate}
    \item \textbf{RPN:} off-the-shelf region proposal network (trained disjoint from this architecture).
    \item \textbf{Crop \& Warp} the patches for fixed network input size
    \item \textbf{Forward} through network for feature map $\rightarrow$ VERY SLOW per-region pass required ($\approx 2000$)
    \item \textbf{Linear Object Classifier} plus a parallel \textbf{Box Regressor} for a refined proposal localization
\end{enumerate}

\paragraph{Fast R-CNN}
\begin{enumerate}
    \item \textbf{FCN transformation} of input into featurized repr. $f_I = f(I)$
    \item unchanged RPN $r(I) \rightarrow$ Region Proposals have VERY POOR RECALL and are slow 
    \item \textbf{RoI Pooling} for a featurized repr of each proposal $g_i = g(f_I,r_i)$: project proposal to feature-map coords (divide by FCN stride $s$), snap fractional coords to integer feature-map cells, split into fixed $k\times k$ grid, max-pool per bin $\rightarrow$ fixed $k\times k\times C$ output regardless of proposal size.
    \item \textbf{ConvNet/MLP} for preprocessing per proposal feature (input already has lower spatial resolution since we use feature maps) $\rightarrow h(g_i)$
    \item \textbf{Multiple Heads} can be attached to these feature maps for task-specific predictions.
\end{enumerate}

\subparagraph{(Multi-Task) Loss}
Since we have two sibling output layers we need to combine their losses for joint training.
\begin{itemize}
    \item $L_{cls} (p,u) = - \log p_u \qquad$ as discrete probability dist per RoI over K+1 categories (CE-loss)
    \item $L_{loc}(t^u,v) = \sum_{i \in \{x,y,w,h\}} \mathrm{smooth}_{L_1}(t^u_i - v_i) \qquad$ summing per-coord robust offsets, with $\mathrm{smooth}_{L_1}(x) = 0.5 x^2$ if $|x|<1$ else $|x| - 0.5$ (Huber: $L_2$ near 0, $L_1$ in tails $\rightarrow$ less outlier-sensitive). Predicts $t^k = (t^k_x, t^k_y, t^k_w, t^k_h)$ for each of the K categories.
    \item \textbf{Combined:} each RoI is given a ground-truth class $u$ and a BB-regression target $v$. The BB-loss is activated for all non-background classes ($\lambda[u\geq 1]$)
    $$
    L(p,u,t^u,v)=L_{cls} (p,u)+\lambda[u\geq 1]L_{loc}(t^u,v)
    $$
\end{itemize}

\paragraph{Faster R-CNN}
\begin{enumerate}
    \item unchanged FCN transformation of input into featurized repr. $f_I = f(I)$
    \item \textbf{Learned Region Proposals} using the already transformed $f_I$ (computation is shared) $\rightarrow RPN(f_I)$:\\
    A $3\times 3$ \textit{sliding window} scans $f_I$; at each position it reads a feature vector that is fed to two sibling heads. At the same position we place $k$ predefined \textit{anchor boxes} (cartesian product of scales $\times$ ratios, e.g. $3\times 3 = 9$). \textbf{Crucially the two boxes are decoupled:} the $3\times 3$ window only defines \textit{where features are read from}; the anchor is the \textit{reference frame for the predictions}. The heads output, \textit{per anchor}:
    \begin{itemize}
        \item \textbf{objectness} $\in [0,1]$: does \textit{this anchor} (not the sliding window) contain an object?
        \item \textbf{box regression} $(dx, dy, dh, dw)$: offsets \textit{relative to the anchor}, not the sliding window.
    \end{itemize}
    So a single feature vector yields $k$ proposals at different scales/ratios sharing one center -- the anchor sets the prior, the deltas fine-tune. Per-location output: $2k$ scores + $4k$ regression values $\rightarrow$ up to $k$ co-located objects detectable per sliding position. Total $H_f W_f k$ proposals, filtered by objectness threshold + NMS before being passed as RoIs.
    \item unchanged RoI Pooling, unchanged ConvNet/MLP, unchanged Multiple Heads
\end{enumerate}
\vspace{-10px}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.8\linewidth]{img/FasterRCNN.png}
    \caption{Faster R-CNN: per-image FCN backbone is shared between the RPN (learned proposals) and the per-region head (RoIPool $\rightarrow$ MLP $\rightarrow$ softmax + box regressor). Training alternates between RPN and detector layers.}
    \label{fig:faster_rcnn}
\end{figure}

\subparagraph{RPN Loss \& Anchor Assignment}
Each anchor $i$ gets a label $p_i^*$:
\begin{itemize}
    \item \textbf{positive} ($p_i^* = 1$): IoU $>0.7$ with any GT box (or the highest-IoU anchor for a GT, to guarantee a match)
    \item \textbf{negative} ($p_i^* = 0$): IoU $<0.3$ with all GT boxes
    \item \textbf{ignored}: in between -- no loss contribution.
\end{itemize}
One GT can match \textit{multiple} anchors. Multi-task loss combines objectness CE and box-regression smooth-$L_1$:
$$
\mathcal{L}(\{p_i\},\{t_i\}) = \frac{1}{N_\text{cls}} \sum_i \mathcal{L}_\text{cls}(p_i, p_i^*) + \lambda \frac{1}{N_\text{reg}} \sum_i p_i^* \mathcal{L}_\text{reg}(t_i, t_i^*)
$$
$\mathcal{L}_\text{cls}$ runs over \textit{all} pos+neg anchors; $\mathcal{L}_\text{reg}$ is gated by $p_i^*$ -- only positives get a regression loss (negatives have no GT box to regress to). Architecturally the sliding window feeds a $256$-d intermediate vector that branches into the $2k$-d cls head and $4k$-d reg head.


\paragraph{Faster R-CNN with Feature Pyramid Network (FPN)}
Rather than using a same scale feature map, we include a multi-scale feature pyramid, i.e., $f_I$ is no longer just one repr at one scale. 
This improves \textit{Scale Invariance} and \textit{Localization Equivariance}.\\

Different Strategies:
\begin{itemize}
    \item \textbf{Image Pyramid}: At each scale predict induvidually - standard, easy but SLOW
    \item \textbf{Multiscale Features (Single-scale Map)}: single feature map by downsampling - fast but suboptimal especially for small objects
    \item \textbf{Naive In-Network Pyramid}: use intermediate features while downsampling for predictions - fast, suboptimal
    \item \textbf{Feature Pyramid Network}: enrich high-res features from low-res features in a top-down manner - fast, less suboptimal, since we use the strong features of low resolution with the fine grained nature of the high resolution
\end{itemize}

\subparagraph{Issues with Indirect Detectors} Anchor-based methods are \textit{indirect}: predictions are made w.r.t. a hand-designed anchor grid rather than the objects themselves. Two pain points:
\begin{itemize}
    \item \textbf{Heuristic GT$\rightarrow$anchor assignment}: IoU thresholds (0.7/0.3), one GT $\to$ multiple anchors -- hyperparameters baked into training.
    \item \textbf{Duplicate predictions}: $k$ anchors per location $\times$ overlapping receptive fields $\rightarrow$ many near-duplicate boxes. Requires \textit{NMS} post-processing at inference time (non-differentiable, extra hyperparameters).
\end{itemize}

\subsubsection{Direct Detectors}
In order to avoid brittle heuristics we can apply direct detectors that drop anchors + NMS entirely. \\
\textbf{Idea:} A direct detector outputs a fixed-size set of $N$ predictions $\hat{y} =
\{\hat{y}_i\}_{i=1}^N$, where $N$ is chosen larger than the max number of objects in any image. No anchors, no
grid; just $N$ slots, each producing $(\hat{p}_i, \hat{b}_i) =$ (class distribution, box). The ground-truth set $y$ is padded with "no object" $\varnothing$ entries up to size $N$ so both sides have the same cardinality.\\
\textbf{Problem:} The network outputs $N$ predicted boxes in arbitrary order. In the indirect methods, we used anchor $\to$ closest GT matching. Here we no longer have these heuristics. 

\paragraph{Direct Set Prediction}
Via \textbf{Bipartite Matching} (NOT training), we can solve the matching as a permutation problem, s.t., $\sigma$ of predictions that pairs each prediction with exactly one GT (or $\varnothing$) so that the total matching cost is minimized:
\begin{align*}
    \hat{\sigma} &= \arg\min_\sigma \sum_i \mathcal{L}_\text{match}(y_i, \hat{y}_{\sigma(i)})\\
    \text{with}\quad y_i &= (c_i,b_i) \quad \hat{y}_{\sigma(i)}=(\hat{p}_{\sigma(i)}, \hat{b}_{\sigma(i)})\\
    \mathcal{L}_\text{match}(y_i, \hat{y}_{\sigma(i)}) &= -[c_i \neq \emptyset] \hat{p}_{\sigma(i)}(c_i)+ [c_i \neq \emptyset]\mathcal{L}_{\text{box}}(b_i, \hat{b}_{\sigma(i)})
\end{align*}
combining both the class probability ($\hat{p}_i$) and the box similarity ($\hat{b}_i$) $\rightarrow$ solved in polynomial time using the \textbf{Hungarian Algorithm}. Note that we use the non-empty-set instances for both the class prob and the box loss. 
We do not care about predictions assigned to $\emptyset$, what we want is ONLY the best matches for the GT. With this we do not punish $\emptyset$-assignments after having found the best GT-matches.\\

On the best match, we than calculate the \textbf{Set Loss}/ Hungarian Loss, where we must tell the network that an unmatched slot should have predicted $\emptyset$ (training), otherwise it would not learn to predict this:
$$
\mathcal{L}_{\text{Hungarian}}(y,\hat{y})=\sum_{i=0}^N-\log \hat{p}_{\hat{\sigma}(i)}(c_i) +[c_i\neq \emptyset]\mathcal{L}_{\text{box}}(b_i,\hat{b}_{\hat{\sigma}(i)}) \quad \text{NLL/CE-loss for strong training signals}
$$

\paragraph{Detection Transformer}
We can use DETR to \textit{generate direct box predictions}.
\vspace{-10px}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.9\linewidth]{img/DETR.png}
    \caption{DETR architecture: CNN backbone $+$ positional encoding $\rightarrow$ transformer encoder over flattened image features $\rightarrow$ transformer decoder cross-attending $N$ learned \textit{object queries} against the encoded features $\rightarrow$ shared FFN heads output $(\text{class}, \text{box})$ per query, with $\varnothing$ for unused slots.}
    \label{fig:detr}
\end{figure}
\begin{itemize}
    \item \textbf{Positional Encoding}: added in the encoder to distinguish same features generated at different spatial locations (transformer is permutation-invariant otherwise).
    \item \textbf{Object Queries}: $N$ \textit{learnt} input embeddings added to the input of each attention layer in the decoder $\rightarrow$ each query becomes one prediction slot.
    \item \textbf{Decoder Self-Attention}: lets queries attend to each other $\rightarrow$ pair-wise relations between objects (the mechanism that suppresses duplicates: queries learn to "divide up" the image).
    \item \textbf{Encoder-Decoder Attention} (cross-attention): each query attends over the full encoded feature map $\rightarrow$ whole image as context per object.
    \item \textbf{Prediction Heads}: FFNs with \textit{shared parameters} but \textit{independent computation per query} $\rightarrow$ $(c_i, b_i)$ per slot.
\end{itemize}

\subsection{Instance Segmentation}
\begin{definition}[Instance Segmentation]
    Object Detection with pixel-level masks instead of bounding boxes for objects. Evaluation still via AP.
\end{definition}

\subsubsection{Mask R-CNN}
We still evolve the R-CNN lineage. The idea is not to \textit{detect-then-segment}. 
\vspace{-10px}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.9\linewidth]{img/MaskRCNN.png}
    \caption{Mask R-CNN: adapted Faster R-CNN with an instance-level segmentation mask head, and RoIAlign.}
    \label{fig:mask_rcnn}
\end{figure}

Core Adaptations of the Mask R-CNN:
\begin{itemize}
    \item \textbf{Instance-Level Segmentation Mask Head}: Decoupled Segmentation from classification such that one mask prediction for each (proposal, class) pair.
    \item \textbf{RoIAlign}: Same concept of transforming an arbitrary-sized proposal into a fixed-dimensional repr, but \textit{without any quantization}.\\
    \textit{Why alignment is needed at all:} the RPN \textit{reads} features from integer feature-map cells but \textit{writes} boxes in continuous image-pixel space (deltas $(dx,dy,dw,dh)\in\mathbb{R}^4$ are continuous on top of the anchor). So when the head re-reads features for that proposal, the box no longer lands on grid cells -- that gap is what RoIPool/RoIAlign bridge.\\
    Recall RoIPool snaps twice -- (i) project RoI to integer feature-map cells, (ii) snap bin boundaries when subdividing -- introducing $\sim$stride-sized misalignments. Fine for classification, but fatal for per-pixel mask prediction. RoIAlign keeps everything fractional:
    \begin{enumerate}
        \item Project the RoI to the feature map -- \textit{do not snap}.
        \item Subdivide into a $k\times k$ grid of bins -- \textit{do not snap} boundaries.
        \item Inside each bin, place a fixed number of \textit{sample points} at exact fractional positions (e.g. uniform $2\times 2$ = 4 samples per bin).
        \item For each sample, get its feature value via \textbf{bilinear interpolation} from the 4 nearest feature-map lattice points (features are treated as living at grid \textit{junctions}, not cell centers).
        \item Bin output = \textit{average} of the 4 interpolated samples.
    \end{enumerate}
    Output is still $k\times k\times C$, but spatially faithful and \textit{differentiable} w.r.t. RoI coordinates (RoIPool is not). Key for masks: $14\times 14$ / $28\times 28$ mask predictions are later resized to the image-space RoI and thresholded at $0.5$, so any RoIPool-style quantization error would propagate directly to mask boundaries.
\end{itemize}

\paragraph{Mask R-CNN Loss}
Builds on the Fast R-CNN multi-task loss by adding a per-pixel mask term. For each proposal the mask head outputs a $K \times m \times m$ feature map $f$ (one $m\times m$ mask per class, no softmax across classes), passed through a sigmoid $\rightarrow$ scores $s$.
\begin{itemize}
    \item $L_\text{mask}(s^u, w) = -\frac{1}{m^2} \sum_{i=1}^{m^2} \left[ w[i] \log s^u[i] + (1 - w[i]) \log(1 - s^u[i]) \right] \qquad$ per-pixel \textit{binary} CE on the sigmoid scores of the mask $s^u$ corresponding to the GT class $u$, against the binary GT mask $w$. Decouples class from mask -- no inter-class competition per pixel (unlike softmax).
    \item \textbf{Combined:} mask loss only on non-background proposals ($[u\geq 1]$):
    $$
    L(p, u, t^u, v, s^u, w) = L_\text{cls}(p, u) + [u\geq 1]\, L_\text{box}(t^u, v) + [u\geq 1]\, L_\text{mask}(s^u, w)
    $$
\end{itemize}


\subsection{Panoptic Segmentation}
\subsubsection{Fundamentals and Definitions}
\begin{definition}[Panoptic Segmentation]
    For a given, pre-defined set of classes $\mathcal{C}$, assign each pixel in the input image individually to a single class. For each pixel, that belong to a countable class, assign it to a unique instance.
\end{definition}

Panoptic Segmentation unifies the instance and classical semantic segmentation into one task, using a uniform output format and a uniform evaluation metric (\textbf{Panoptic Quality}).\\
\textbf{Data} is in generic supervised learning form: $\mathcal{D} = \{(x_i,y_i): i = 1, \dots, N\}$ . Thus, with $Z$ as the max number of instances for a given class: 
\begin{align*}
    \text{Input (RGB):}& \quad x_i \mathrel{\hat{=}} \mathbf{I}_i \in \mathbb{R}^{H\times W \times 3}\\
    \text{Output:} &\quad \mathbf{Y}_i \in \{1,\dots, C\}\times \{1, \dots, Z\}^{H \times W}\\
    \text{Loss:}& \text{ depending on the architecture (instance-semantic or unified mask-based)}
\end{align*}

For each pixel $i$: \texttt{(class, instance count (if countable))}. Contrary to instance segmentation we have \textbf{no prediction overlaps} (but arbitrary instance ordering), and \textbf{no instance confidence scores}.\\


\paragraph{Panoptic Quality (PQ).}
\textbf{Fundamental Principle:} Each GT segment can have at most one corresponding predicted segment with $IoU > 0.5$ and vice versa. Thus matching is unique and finding it is simply \textit{the first match with IoU $> 0.5$}.\\
$$
\text{PQ} = \frac{\sum_{(p,g) \in TP} \text{IoU}(p,g)}{|TP| + \tfrac{1}{2}|FP| + \tfrac{1}{2}|FN|} \qquad \text{for a single class}
$$

We can interpret this as a product of segmentation quality (SQ) and recognition quality (RQ):
$$
\text{PQ} = \underbrace{\frac{\sum_{(p,g) \in TP} \text{IoU}(p,g)}{|TP|}}_{\text{SQ}} \times \underbrace{\frac{|TP|}{|TP| + \tfrac{1}{2}|FP| + \tfrac{1}{2}|FN|}}_{\text{RQ}}
$$
\begin{itemize}
    \item \textbf{RQ} = F1 score (harmonic mean of precision \& recall) -- detection quality.
    \item \textbf{SQ} = average IoU over matched (TP) segments -- localization/mask quality.
    \item SQ depends on RQ (only the matched segments enter SQ).
    \item Decomposition only holds \textit{per class} -- does not hold for mean PQ across classes.
\end{itemize}




\subsubsection{Instance-semantic Framework for Panoptic Segmentation}
Idea: re-use existing instance and semantic segmenters to obtain a panoptic output without designing a new model $\rightarrow$ combine their predictions post-hoc.

\paragraph{Baseline: Merge Independent Predictions}
Run an instance segmenter (Mask R-CNN, things only) and a semantic segmenter (stuff + things) independently, then merge pixel-wise: things win over stuff; overlapping instances resolved by confidence.
\begin{itemize}
    \item PQ$^\text{Th}$ unchanged (instances taken as-is); PQ$^\text{St}$ drops slightly (instance masks carve into stuff).
    \item \textbf{Failure mode}: \textit{unlabeled (black) halos around boundaries} -- semantic seg predicts a thing class but no instance mask covers the pixel $\rightarrow$ no valid instance ID $\rightarrow$ pixel unlabeled. Structural flaw of independent models.
    \item Motivates \textit{backbone-sharing} models (Panoptic FPN).
\end{itemize}

\paragraph{Panoptic FPN: Shared Backbone}
\textbf{Motivation}: the baseline fails structurally because the two segmenters never share features $\rightarrow$ no way to agree on boundaries (black halos). Sharing a backbone (a) forces the features to be jointly useful for both tasks, (b) is essentially free since FPN already produces \textit{high-resolution, rich, multi-scale} features -- exactly the standard requirements for semantic seg.\\
Starting point: FPN-based Mask R-CNN. Idea: add a \textit{semantic segmentation branch} fed by the same FPN $\rightarrow$ instance + semantic heads jointly trained on a shared feature pyramid.
\begin{itemize}
    \item \textbf{Semantic branch}: each FPN level (resolutions $1/32, 1/16, 1/8, 1/4$ with $256$ channels) is upsampled to a common $1/4$ resolution via a stack of \textit{$\text{conv}\to 2\times$ upsample} blocks (more blocks for coarser levels) producing $128 \times 1/4$ maps, element-wise summed, then $\text{conv}\to 4\times$ to image resolution $\rightarrow$ per-pixel $C$-way logits.
    \item \textbf{Training loss}: weighted sum to account for the different ranges of the two losses (instance heads are per-RoI, semantic head is per-pixel over the full image):
    $L = \lambda_\text{inst}(L_\text{cls} + L_\text{box} + L_\text{mask}) + \lambda_\text{sem} L_\text{sem}$
    \item \textbf{Panoptic inference} (heuristic merge -- panoptic format forbids overlap):
    \begin{itemize}
        \item instance-instance overlap $\rightarrow$ higher confidence wins;
        \item instance-stuff overlap $\rightarrow$ instance wins;
        \item stuff segments below an area threshold are dropped.
    \end{itemize}
    \item \textbf{Limitation}: heads are still \textit{segregated} -- they share a backbone but make independent predictions reconciled by hand-crafted rules $\rightarrow$ suboptimal post-processing. Motivates the unified mask-based formulation below.
\end{itemize}

\subsubsection{Unified Mask-based Framework for Segmentation}
Rather than disjointly learning instance and semantic segmentation, we combine them into one unified objective.

\paragraph{Segmentation as Mask-based Classification}
Both instance and semantic seg can be written as predicting a \textit{set} of \textit{(mask, class)} pairs. This lets us reuse DETR's set-prediction recipe, only swapping the geometric output from a \textit{box} to a \textit{mask}.
\begin{itemize}
    \item \textbf{Ground truth}: $z^\text{gt} = \{(c_i^\text{gt}, m_i^\text{gt})\}_{i=1}^{N^\text{gt}}$ with $c_i^\text{gt} \in \{1,\dots,K\}$, $m_i^\text{gt} \in \{0,1\}^{H\times W}$.
    \item \textbf{Prediction}: $z = \{(p_i, m_i)\}_{i=1}^N$ -- $N$ slots, each a class \textit{distribution} $p_i$ and a \textit{soft} mask $m_i \in [0,1]^{H\times W}$.
    \item \textbf{Matching}: pad $z^\text{gt}$ with $\varnothing$ tokens up to $N$, then bipartite-match with the Hungarian algorithm (cf.\ DETR).
\end{itemize}
\textbf{Matching cost} per pair (both terms gated on $c_i^\text{gt}\neq \varnothing$; raw $p$ -- no log -- so the two terms stay on the same numerical scale and neither dominates the matching, same logic as DETR):
$$\mathcal{L}_\text{match}(z_{\sigma(i)}, z_i^\text{gt}) = -[c_i^\text{gt}\neq\varnothing]\, p_{\sigma(i)}(c_i^\text{gt}) + [c_i^\text{gt}\neq\varnothing]\, \mathcal{L}_\text{mask}(m_{\sigma(i)}, m_i^\text{gt})$$
$\mathcal{L}_\text{mask}$: binary mask overlap loss (details later).\\

\textbf{Hungarian / set loss} (a.k.a.\ \textit{mask-classification loss}) given the optimal match $\hat\sigma$:
$$\mathcal{L}_\text{mask-cls}(z, z^\text{gt}) = \sum_{i=1}^N -\log p_{\hat\sigma(i)}(c_i^\text{gt}) + [c_i^\text{gt}\neq\varnothing]\, \mathcal{L}_\text{mask}(m_{\hat\sigma(i)}, m_i^\text{gt})$$
Same DETR-style asymmetry: the cls term is \textit{not} gated by $[c_i^\text{gt}\neq\varnothing]$ -- unmatched slots are explicitly supervised toward the $\varnothing$ class to suppress duplicates; switching to $-\log p$ gives proper NLL gradients during training. Mask term stays gated (no GT mask for $\varnothing$ slots).

\paragraph{MaskFormer}
Similar idea as DETR but \textit{without a transformer encoder} -- segmentation needs per-pixel features at near-image resolution, and self-attention is $\mathcal{O}(N^2)$ in tokens, so flattening high-res feature maps into an encoder would be prohibitive. Instead a \textit{pixel decoder} (FPN-style CNN upsampling) produces the dense per-pixel features that the transformer decoder's queries cross-attend against.
\begin{figure}[H]
    \centering
    \includegraphics[width=\linewidth]{img/MaskFormer.png}
    \caption{Mask Former High Level View}
    \label{fig:maskformer}
\end{figure}

\subparagraph{Mask Prediction} Sigmoid dot product of per-query \textit{mask embedding} $\mathcal{E}_\text{mask}[:,i]$ with per-pixel embedding $\mathcal{E}_\text{pixel}[:,h,w]$ from the pixel decoder:
$m_i[h,w] = \sigma(\mathcal{E}_\text{mask}[:,i]^\top \cdot \mathcal{E}_\text{pixel}[:,h,w])$
Interpreted as the \textit{non-exclusive degree} to which pixel $(h,w)$ belongs to mask $i$ -- no softmax across slots, so different slots' masks can overlap. The panoptic no-overlap constraint is enforced only at inference (cf.\ Mask R-CNN's per-class masks, here per-query).

\subparagraph{Same Architecture, Different Weights} One architecture for semantic / instance / panoptic, but a \textit{separate model is trained per task} -- unified \textit{formulation}, not unified \textit{weights} (Mask2Former goes further).

\subparagraph{Inference} Training is fixed (mask-cls loss); only the inference rule switches per task:
\begin{itemize}
    \item \textbf{Panoptic / instance}: per pixel, pick the \textit{dominant} (slot, class) pair, $\arg\max_{i\,:\,c_i\neq\varnothing} p_i(c_i)\cdot m_i[h,w]$ -- both class- and mask-conf must be high. Each instance stays a distinct segment.
    \item \textbf{Semantic}: \textit{marginalize} over slots, $\arg\max_{c\in\{1,\dots,K\}} \sum_{i=1}^N p_i(c)\cdot m_i[h,w]$. Semantic only cares about the class per pixel, not which slot produced it $\rightarrow$ multiple "person" slots can pool their evidence into one "person" region.
    \item Training the semantic objective directly would scatter the gradient across all $N$ queries (each gets only a fraction of the signal) $\rightarrow$ keep mask-cls loss, switch inference only.
    \item However, specialized methods require siginificantly fewer epochs for equivalent performance
\end{itemize}

\paragraph{Mask2Former}
Same fundamental idea as MaskFormer but with significant improvements:
\begin{itemize}
    \item \textbf{Masked Attention} (in cross-attention): restrict each query to attend \textit{only to pixels inside its current mask prediction}, by adding an attention bias
    $\mathcal{M}_{l-1}(x,y) = \begin{cases} 0 & \text{if } M_{l-1}(x,y) = 1 \\ -\infty & \text{otherwise} \end{cases}$
    inside the softmax: $X_l = \text{softmax}(\mathcal{M}_{l-1} + Q_l K_l^\top)V_l + X_{l-1}$. $-\infty$ entries vanish after softmax $\rightarrow$ those pixels contribute zero. Intuition: \textit{global context takes time to learn}; for cross-attention local foreground features already suffice, broader context is gathered by self-attention.
    \item \textbf{Swap Self- and Cross-Attention}: queries at the first decoder layer are image-independent (purely learned embeddings) $\rightarrow$ self-attention among them carries no image content to share. Do cross-attention first to inject image features into queries, \textit{then} self-attention -- opposite of the transformer default.
    \item \textbf{High-Res Features from Pixel Decoder}: MaskFormer decoded queries against the \textit{low-res bottleneck} features only; Mask2Former feeds each successive transformer-decoder layer a \textit{different scale} from the pixel decoder (multi-scale). Helps segment objects of diverse scales, especially small ones.
    \item \textbf{Point Sampling} (matching cost \& mask-cls loss approximation): $\mathcal{L}_\text{mask}$ over the full $H\times W$ map for every (prediction, GT) pair is memory-heavy. Approximate it on $K$ sampled pixels:
    \begin{itemize}
        \item \textit{Matching cost}: \textit{uniform} sampling of a \textit{shared} pixel set across all pairs $\rightarrow$ costs stay comparable across pairs (apples-to-apples for the Hungarian).
        \item \textit{Mask-cls loss}: \textit{importance} sampling of \textit{distinct} pixel sets per mask $\rightarrow$ each mask gets its own boundary/hard-pixel focus.
    \end{itemize}
    Drops the full-image mask compute requirement $\rightarrow$ enables training on standard hardware.
\end{itemize}

\subsection{Depth and 3D Estimation}
\textit{Stereo vision} to generate 3D renderings from 2D images. Other possibilities include: \textit{structured light, ultrasound, radar, time of flight (active), and cam/lidar combinations}.
We can deduce some information also from 2D images using cues including shading, texture, focus, and motion. \\

Challenges to monocular depth estimation: a single 2D image may be produced from an infinite number of 3D scenes due to \textbf{scale ambiguity} $\rightarrow$ overcome some of this by using cues or features such as perspective or texture infromation, object sizes, locations and occlusions. \\

$\rightarrow$ regression problem (cont output per pixel), thus we use MSE or vaiants. (no longer MAP!)

\paragraph{Ground Truth.}
\begin{itemize}
    \item \textbf{LiDAR} (e.g.\ Velodyne): sparse; limited vertical FoV; errors on moving objects and on specular/transparent surfaces. In practice can be \textit{densified} for static parts by merging consecutive frames (imperfect, but less sparse).
    \item \textbf{Structured light} (e.g.\ Kinect): fails outdoors (sunlight interference); short range ($\sim$few meters); errors on specular/transparent surfaces.
\end{itemize}

\paragraph{Evaluation Metrics}
Comparing predicted depth map $Y$ to GT $Y^*$ over $T$ valid depth pixels.
\begin{itemize}
    \item \textbf{Threshold accuracy}: fraction of pixels with $\max\!\left(\frac{y_i}{y_i^*}, \frac{y_i^*}{y_i}\right) < \text{thr}$ (symmetric ratio $\geq 1$ $\rightarrow$ penalizes over- and under-estimation). Counts correct pixel given some tolerance; typical threshold $1.25$. Robustness-flavored.
    \item \textbf{Absolute relative difference}: $\text{rel} = \frac{1}{T}\sum_{i,j}|y_{i,j} - y_{i,j}^*|/y_{i,j}^*$ -- per-pixel error normalized by GT depth (far pixels not over-weighted).
    \item \textbf{Squared relative difference}: $\text{srel} = \frac{1}{T}\sum_{i,j}|y_{i,j} - y_{i,j}^*|^2/y_{i,j}^*$ -- penalizes larger errors more.
    \item \textbf{RMS (linear)}: $\sqrt{\frac{1}{T}\sum_{i,j}|y_{i,j} - y_{i,j}^*|^2}$ -- standard L2; dominated by far/large-depth errors.
    \item \textbf{RMS (log)}: $\sqrt{\frac{1}{T}\sum_{i,j}|\log y_{i,j} - \log y_{i,j}^*|^2}$ -- compares depth \textit{magnitudes} (orders of magnitude); equalizes near/far contributions.
\end{itemize}

\subsubsection{Supervised Methods}
Standard CNN based Enc-Dec methods with loss between output and GT depth.

\textbf{Loss:} While \textit{L1} and \textit{L2} are possible, the \textbf{berHu loss} is more common:
$\mathcal{B}(x) = \begin{cases} |x| & |x| \leq c \\ \frac{x^2 + c^2}{2c} & |x| > c \end{cases}$\\
\textit{Inverse of Huber}: L1 for small residuals, L2 for large ones. \\

Computed as $\mathcal{B}(\tilde y - y)$ with $c = \tfrac{1}{5}\max_i(|\tilde y_i - y_i|)$ over all pixels in the current batch -- the L1/L2 cut-off is set to $20\%$ of the largest residual in the batch (adaptive scale).

\paragraph{P3Depth.}
Leveraging regularities in real-world (planar) scenes yielding piecewise-smooth depth maps. $\rightarrow$ rather than directly predicting depth $D$, predict a per-pixel repr $\mathbf{C}$ that is \textit{identical for pixels on the same 3D plane}, then use neighbor information in $\mathbf{C}$ to predict the depth of each pixel.

\subparagraph{Plane Coefficient Representation}
Backproject a pixel $\mathbf{p} = (u,v)^\top$ to a 3D point $\mathbf{P} = (X,Y,Z)^\top$ via the pinhole model:
$Z = D(u,v), \quad X = \tfrac{Z(u-u_0)}{f_x}, \quad Y = \tfrac{Z(v-v_0)}{f_y}$
A 3D plane with normal $\mathbf{n} = (a,b,c)^\top$ and offset $d$ satisfies $\mathbf{n}\cdot\mathbf{P} + d = 0$. Substituting the backprojection and rearranging gives a \textit{linear function for $1/Z$ in pixel coords}:
$$\tfrac{1}{Z} = \underbrace{\tfrac{-a}{f_x d}}_{\hat\alpha} u + \underbrace{\tfrac{-b}{f_y d}}_{\hat\beta} v + \underbrace{\tfrac{1}{d}\!\left(\tfrac{a}{f_x}u_0 + \tfrac{b}{f_y}v_0 - c\right)}_{\hat\gamma}\qquad \text{Normalize:} \rho = \sqrt{\hat\alpha^2 + \hat\beta^2 + \hat\gamma^2}\quad (\alpha,\beta,\gamma) = (\hat\alpha, \hat\beta, \hat\gamma)/\rho$$ 
$\rightarrow \textbf{plane coefficients} \mathbf{C} = (\alpha, \beta, \gamma, \rho)^\top$.
Depth recovered from $\mathbf{C}$ and pixel coords:
$Z = [(\alpha u + \beta v + \gamma)\rho]^{-1} = h(\mathbf{C}, u, v)$
Pixels on the same 3D plane share the same $\mathbf{C}$ -- it depends only on camera intrinsics and the plane, not the individual pixel $\rightarrow$ enables neighborhood-based depth refinement.

\subparagraph{Injecting Regularities}
In order to acually use these \textbf{interaction between pixels in the new output space}, we must fuse the spaces. Simply changing hte output space to a repr more closely related to 3D planes does NOT inject these priors.
\vspace{-0.3cm}
\begin{figure}[H]
    \centering
    \includegraphics[width=\linewidth]{img/P3Depth.png}
    \label{fig:P3Depth}
\end{figure}
\vspace{-0.5cm}
Two decoders branch off a shared feature extractor:
\begin{itemize}
    \item \textbf{Plane Coefficient Decoder} $\rightarrow$ dense plane-coefficient map $\mathbf{C}(u,v) = (\alpha, \beta, \gamma, \rho)$.
    \item \textbf{Offset Vector Field Decoder} $\rightarrow$ per-pixel offset $\mathbf{o}(\mathbf{p}) = \mathbf{q} - \mathbf{p}$ pointing to a \textbf{seed pixel} $\mathbf{q}$ in the neighborhood that is associated with the \textit{same 3D plane} as $\mathbf{p}$.
\end{itemize}
Two depth predictions are then computed:
\begin{itemize}
    \item \textbf{Initial}: $D_i(u,v) = h(\mathbf{C}(u,v), u, v)$ -- each pixel uses its own coefficients.
    \item \textbf{Seed-based}: resample coefficients at the seed location, $\mathbf{C}_s(\mathbf{p}) = \mathbf{C}(\mathbf{p} + \mathbf{o}(\mathbf{p}))$, then $D_s(u,v) = h(\mathbf{C}_s(u,v), u, v)$ -- uses the seed's plane params but $\mathbf{p}$'s pixel coords (valid since both share the same plane).
\end{itemize}
\textit{Why the seed at all?} CNN/transformer decoders produce smooth feature maps $\rightarrow$ plane coefficients are reliable in plane \textit{interiors} but blur near plane boundaries. The offset snaps each pixel to a more reliable interior seed of its own plane $\rightarrow$ sharp depth discontinuities at edges.\\

\textbf{Adaptive Fusion}: $D_f(u,v) = F(u,v) D_s(u,v) + (1 - F(u,v)) D_i(u,v)$, with predicted confidence $F\in[0,1]$. High $F$ trusts the piecewise-planar prior; low $F$ falls back to direct prediction for non-planar surfaces (curved, high-texture).\\
\textbf{Training}: supervised \textit{only on depth}: $\mathcal{L}(D_i, D^*) + \mathcal{L}(D_s, D^*) + \mathcal{L}(D_f, D^*)$. Offsets, plane coefficients, and confidence are all learned \textit{implicitly} -- no GT for them.

\subsubsection{Stereo Depth Estimation}
Using a stereo setup we can estimate depth using triangulation. \\

We consider \textit{identical camers, with coplanar image planes, aligned x-axes and the convention of setting the left camera frame to the world coords.} Thus have a translation along the known baseline (in meters) and no rotation.
\vspace{-0.4cm}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.8\linewidth]{img/Stereo_Setup.png}
    \label{fig:StereoSetup}
\end{figure}

The translation from the $C_L = W$ to $C_R$ is then given by:
$$
\begin{pmatrix} x^r \\ y^r \\ z^r \\ 1 \end{pmatrix} = \begin{pmatrix} \mathbf{R} & \mathbf{t} \\
  \mathbf{0}^\top & 1 \end{pmatrix} \begin{pmatrix} x^w \\ y^w \\ z^w \\ 1 \end{pmatrix} \qquad \text{with} \quad
  \mathbf{R} = \mathbf{I} \text{ and } \mathbf{C} = (b\ 0\ 0)^T = -t
$$

\subparagraph{Projection \& Disparity}
Further assume image coords already centered at the principal point ($(u_0,v_0) = (0,0)$) and $K_L = K_R = K$ with $K = \text{diag}(kf, lf, 1)$. Projecting the world point through each camera:
$\rho\begin{pmatrix}u\\v\\1\end{pmatrix} = K\begin{pmatrix}x^w\\y^w\\z^w\end{pmatrix}, \qquad \rho'\begin{pmatrix}u'\\v'\\1\end{pmatrix} = K\begin{pmatrix}x^w - b\\y^w\\z^w\end{pmatrix}$
Dehomogenizing:
$$u = \tfrac{kfx^w}{z^w},\; v = \tfrac{lfy^w}{z^w}, \qquad u' = \tfrac{kf(x^w - b)}{z^w},\; v' = \tfrac{lfy^w}{z^w}$$
$\rightarrow v = v'$: rectified pair, the same scene point lies on the same image row in both cameras (only horizontal disparity).\\

\textbf{Disparity}: $d = u - u'$ -- the horizontal displacement of the same point between the two images. Solving the equations above for the 3D coords:
$$x^w = b\tfrac{u}{u - u'}, \qquad y^w = b\tfrac{k}{l}\tfrac{v}{u - u'}, \qquad z^w = bk\tfrac{f}{u - u'}$$
Depth is inversely proportional to disparity. Normally $k = l$ (square pixels). Near objects $\rightarrow$ large $d$ (well-resolved); far objects $\rightarrow$ small $d$ (precision degrades quadratically with distance).

\subparagraph{Visible Region}
\begin{itemize}
    \item \textbf{Blind region}: zone in front of the cameras visible to only one (or neither) $\rightarrow$ no stereo depth possible.
    \item \textbf{Valid stereo region}: intersection of both FoVs $\rightarrow$ correspondences (hence depth) computable.
\end{itemize}

\subparagraph{Design Trade-offs \& Remarks}
\begin{itemize}
    \item Increasing $b$ and/or $f$ $\rightarrow$ better depth resolution, but smaller simultaneous visibility (larger blind region; narrower FoV for $f$).
    \item \textit{Same disparity = same depth} holds \textit{only for this rectified simple setup} -- not for general stereo geometries.
    \item Too-small disparity $\rightarrow$ depth becomes unresolvable: sub-pixel matching noise translates to large depth ambiguities (quadratic in distance) $\rightarrow$ far objects unreliable.
    \item The real bottleneck is \textbf{finding correspondences} under non-ideal inputs: occlusions, repetitive patterns, textureless regions, reflective surfaces.
\end{itemize}

\subparagraph{Stereo Matching Algorithm and Siamese Networks.}
Classical four-step process: (I) matching cost, (II) cost aggregation, (III) optimization, (IV) disparity refinement. \textbf{Siamese Networks} = two twin networks with \textit{shared parameters} processing two inputs in parallel, trained to minimize the distance for matching pairs and maximize it for distinct ones $\rightarrow$ natural fit for step (I).

\begin{figure}[H]
    \centering
    \includegraphics[width=0.8\linewidth]{img/Sieamese.png}
    \label{fig:deepstereoregression}
\end{figure}

\textbf{End-to-end deep stereo regression} replaces all four steps with learned blocks:
\begin{itemize}
    \item \textbf{Siamese 2D conv} -- shared-weight feature extractor on both images $\rightarrow$ learned per-pixel descriptors (replaces hand-crafted matching cost of step I).
    \item \textbf{Cost volume} (a.k.a.\ \textit{correlation layer}) -- \textit{no learnable parameters}; a fixed construction that for every $(u,v)$ and candidate disparity $d$ combines $f_L(u,v)$ with $f_R(u-d, v)$ -- by concatenation ($H\times W\times (D_{\max}+1)\times 2C$) or dot product ($H\times W\times (D_{\max}+1)$). Built in \textit{both directions} (L$\rightarrow$R and R$\rightarrow$L) $\rightarrow$ two cost volumes, one anchored per image, enabling downstream left-right consistency / occlusion reasoning. Still differentiable $\rightarrow$ gradients flow back into the Siamese features. It is the \textit{inductive bias} of the architecture (only same-row, fixed-shift pairs are considered).
    \item \textbf{Multi-scale 3D conv / 3D deconv} -- propagates context across spatial \textit{and} disparity dimensions $\rightarrow$ cost aggregation + optimization (steps II+III). \textit{Why 3D conv, not 2D?} 2D conv would slice across $(H,W)$ per disparity independently, decimating the disparity axis; 3D conv treats $D$ as a proper spatial dim and preserves it. Geometry-motivated design, not a black box. Output: a refined cost volume reduced to a \textit{single feature channel} -- $H\times W\times (D_{\max}+1)$ scalar costs.
    \item \textbf{Soft argmax / soft argmin} -- differentiable disparity selection (step III). A hard argmin over costs would (i) only allow integer disparities $\rightarrow$ quantized depth and (ii) make the loss \textit{fragile}: a prediction one off from GT incurs the same loss as one completely off (no gradient pointing toward correct). Instead treat costs as negated logits, softmax over the disparity axis, take the convex combination:
    $\hat d[i,j] = \sum_{D=0}^{D_{\max}} D \cdot \mathrm{softmax}_D(-C[i,j,D])$
    $\rightarrow$ continuous, sub-pixel disparity that flows gradients smoothly. Final loss: $L_1$ regression 
    $$L = \tfrac{1}{HW}\sum_{i,j} \lVert \hat d[i,j] - d[i,j] \rVert_1$$.
\end{itemize}
\vspace{-0.5cm}
\textit{Why a cost volume rather than a naive concat?} A naive baseline would just concatenate the left and right feature maps along channels and regress disparity with a big CNN -- but this throws away the stereo geometry, forcing the network to \textit{discover} that matches lie along the same row at a small horizontal shift. The cost volume bakes that prior in: each disparity slice $d$ holds the similarity between the left feature at $(u,v)$ and the right feature at $(u-d, v)$ $\rightarrow$ only same-row, fixed-shift correspondences are considered, so the network just has to \textit{pick} the best disparity.

\subsubsection{Self-Supervised Monocular Depth Estimation from Stereo}
\textbf{Key asymmetry}: stereo is used at \textit{training} time as a supervision signal; the CNN only ever sees the \textit{left} image.
\begin{itemize}
    \item \textbf{Forward pass}: $\text{CNN}(L) \rightarrow$ disparity map.
    \item \textbf{Loss}: warp $R$ using the predicted disparity (via the stereo projection equations $u = \tfrac{kfx^w}{z^w}, u' = \tfrac{kf(x^w-b)}{z^w}$) $\rightarrow$ synthesized $\tilde L$; photometric loss $\lVert \tilde L - L\rVert$. Right image is supervision only, not input.
    \item \textbf{Why "monocular"}: at test time only one camera is needed. Stereo provides cheap, scalable self-supervision
    \item \textbf{Accuracy}: not as good as fully supervised, but decent.
    \item \textbf{Issues}: occlusions, FoV mismatch between cameras (some pixels in $L$ have no valid match in $R$).
\end{itemize}

\subsubsection{Self-Supervised Depth and Motion Estimation from Video}
Extends stereo self-supervision to \textit{monocular video}: no known baseline, so the camera motion between consecutive frames must also be predicted. Train on raw video, deploy on a single image.
\vspace{-0.3cm}

\begin{figure}[H]
    \centering
    \includegraphics[width=0.8\linewidth]{img/DepthAndMotion.png}
    \caption{Unsupervised Learning of Depth and Ego-Motion from Video}
    \label{fig:depthandmotion}
\end{figure}
\vspace{-0.3cm}


Two jointly trained networks:
\begin{itemize}
    \item \textbf{Depth CNN}: $I_t \rightarrow \hat D_t(p)$ -- per-pixel depth of the target frame.
    \item \textbf{Pose CNN}: $(I_t, I_{t\pm 1}) \rightarrow \hat T_{t\rightarrow s}$ -- relative camera pose (\textit{ego-motion}) between target and source frames.
\end{itemize}
\textbf{Warping equation}: for a pixel $p_t$ in the target, the corresponding source pixel is
$p_s \sim K\, \hat T_{t\rightarrow s}\, \hat D_t(p_t)\, K^{-1} p_t$
(unproject $p_t$ via $K^{-1}$, scale by depth, transform by pose, reproject via $K$). The source image is then sampled at $p_s$ via \textit{bilinear interpolation} (differentiable w.r.t. warp $\rightarrow$ gradients reach both networks).\\
\textbf{View-synthesis photoconsistency loss}: $\mathcal{L}_{vs} = \sum_s \sum_p |I_t(p) - \hat I_s(p)|$ -- L1 between target and warped source frame(s). Depth and pose are supervised \textit{only} through this signal.\\
\textbf{Failure modes}: breaks on specular / non-Lambertian surfaces (photometric assumption violated); moving objects violate the static-scene assumption $\rightarrow$ commonly worked around by global motion params or learned masks.

\subsection{3D Classification, Segmentation, and Detection}
\subsubsection{Foundations and Definitions}
\textbf{Representations:} many possibilities including multi-view, volumetric (voxels), part assembly, \textbf{Point Clouds}, mesh (graph CNN), and implicit shapes (functional repr.) $\rightarrow$ focus on point clouds since close to raw sensor data and common standard. Point Clouds can either be converted into other reprs to feed into NNs or we can do \textit{feature learning} directly on point clouds (our focus)\\
\textbf{3D Perception Tasks}: Classification, Segmentation, Detection, Correspondences, Reconstruction

\subsubsection{3D Classification and Semantic Segmentation on Point Clouds}
\paragraph{PointNet.}
Point Cloud based end-to-end learning for \textit{scattered, unordered, sparse, noisy} point data for various tasks. Challenges caused by the data foundation:
\begin{itemize}
    \item \textbf{Unordered} input point set: Model requires invariance to $\text{N}!$ permutations (N orderless points). This can be achieved via \textit{Symmetric Functions} (c.f. GNNs): Trivial Solution is the \textit{MLP with max pooling} (across features): $f(x_1, \dots, x_n) = \gamma \circ g(h(x_1), \dots, h(x_n))$ symmetric if $g$ is symmetric:\\
    $\underbrace{x}_{\text{input}} \rightarrow \underbrace{\texttt{MLP}(x_i)}_{h} \rightarrow \underbrace{\texttt{max\_pool}}_{g} \rightarrow \underbrace{\texttt{MLP}}_{\gamma} \qquad $ the symmetric $g$ (max-pool) is what ensures permutation invariance; $\gamma$ just maps the aggregated feature to the output
    \item \textbf{Geometric Transformation Invariance}: rotations / Euclidean transformations must not alter results. Solution: \textbf{T-Net} -- a mini vanilla PointNet whose output is a transformation \textit{matrix} (not a class score), applied by matrix-mult to align inputs to a \textit{canonical space}. Used twice: $3\times 3$ on raw $n\times 3$ points (\textit{input} T-Net), $64\times 64$ on intermediate $n\times 64$ features (\textit{feature} T-Net). Data-dependent $\rightarrow$ no hard-coded rotations.
\end{itemize}

\subparagraph{PointNet Limitations}
\begin{itemize}
    \item \textbf{No local context per point}: features are either \textit{per-point} (one point seen at a time) or \textit{global} (all points aggregated by max-pool) -- nothing in between. Contrast with 3D CNNs that have hierarchical feature learning across multiple receptive-field scales.
    \item \textbf{Global feature depends on absolute coordinates}: MLPs ingest raw $(x,y,z)$, so the global descriptor is tied to absolute position $\rightarrow$ same object at a different location yields a different feature $\rightarrow$ hard to generalize to unseen scene configurations.
\end{itemize}

\paragraph{PointNet++}
Addresses local context per point and global feature dependence on absolute coordinates via hierarchical local-neighborhood processing. The goal is to \textit{mimic convolution} style neighborhood processing. \\
In scattered point clouds, however, we cannot apply simple convolution, thus we use one of the following to find neighbors:
\begin{itemize}
    \item Ball Query: fix a radius $r$, take all points within. Fixed metric "kernel" $\rightarrow$ more stable across density variations (the receptive field is the same scale everywhere). Variable number of neighbors.
    \item K-NN Query: distance problematic, since $K$ can represent very distant context if static.
\end{itemize}

\subparagraph{Set Abstraction Levels} 
Hierarchical structure = a stack of \textit{set abstraction} levels, each repeating three steps:
\begin{enumerate}
    \item \textbf{Sampling Layer:} \textit{Iterative Farthest Point Sampling} (FPS) -- given the current sampled set $S$, the next point is $x_\text{next} = \arg\max_{x \notin S} \min_{s \in S} \lVert x - s\rVert$ (the unsampled point whose \textit{nearest sampled neighbor} is farthest) $\rightarrow$ each new point fills the largest gap $\rightarrow$ \textit{better coverage} than random sampling.
    \item \textbf{Grouping Layer:} For each centroid, run ball query (or K-NN) $\rightarrow$ output groups $N' \times K \times (d+C)$. $K$ varies per group (ball query is radius-based); the next PointNet collapses the variable-$K$ axis to a fixed-length feature via max-pool.
    \item \textbf{PointNet Layer:} Apply PointNet inside each group, taking points \textit{relative} to the centroid $\rightarrow$ one feature per centroid (the analog of one convolution output).
\end{enumerate}
Stacking: $N$ points $(x,y) \rightarrow N_1$ anchors $(x,y,\mathbf{f}) \rightarrow N_2$ anchors $(x,y,\mathbf{f}')\rightarrow \dots$ -- fewer points, richer features per level (CNN-style downsampling). Two limitations of vanilla PointNet are fixed: (i) each anchor's feature now \textit{sees its local neighborhood} $\rightarrow$ hierarchical local context; (ii) since inner PointNets see \textit{relative} coordinates, the features are translation-invariant.

\begin{figure}[H]
    \centering
    \includegraphics[width=\linewidth]{img/PointNet++.png}
    \caption{PointNet++ Architecture}
    \label{fig:point-net++}
\end{figure}
\vspace{-0.3cm}

\textit{Feature Interpolation for PointNet++ Decoder:} For per-point tasks (segmentation) we need to undo the set-abstraction subsampling and propagate features back to denser resolution. For each output point $\mathbf{x}_j$, find its $k$ nearest neighbors $\{\mathbf{x}_i\}$ in the (subsampled) input set, then compute an inverse-distance-weighted average:
$$w_{ij} = \frac{1}{d(\mathbf{x}_j, \mathbf{x}_i)^p}, \qquad \mathbf{f}(\mathbf{x}_j) = \frac{\sum_{i=1}^k w_{ij}\, \mathbf{f}(\mathbf{x}_i)}{\sum_{i=1}^k w_{ij}}$$
Closer subsampled points contribute more; typically $k=3$, $p=2$. Density-aware upsampling -- the decoder analog of bilinear upsampling in 2D CNNs.

\subsubsection{3D Object Detection on Point Clouds}
\textbf{Target}: 3D BB around each object given \textit{3 DOF for size, 3 DOF for center, and 1 DOF for rotation} - assuming object on ground (no vertical rotation).\\

\textbf{LiDAR} provides reliable depth information used to localize and characterize objects. However, LiDAR point clouds are \textit{sparse, irregular and have highly variable point density}. We can adress these challenges by \textit{manually crafting feature repr} for point clouds, \textit{point cloud projection into perspective view} for image-based techniques, or \textit{rasterize point clouds into 3D voxel grids} and encode each voxel with hand-crafted or PointNet features.

\paragraph{VoxelNet}
While a pure 2D inspired sliding window and classification approach is theoretically possible, the search space is huge (cubic increase with resolution).\\

$\rightarrow$ two-stage approach \textbf{object proposal + refinement}
\vspace{-0.3cm}
\begin{figure}[H]
    \centering
    \includegraphics[width=\linewidth]{img/VoxelNet.png}
    \caption{VoxelNet Architecture with Feature Learning Network Details}
    \label{fig:VoxelNet}
\end{figure}
\vspace{-0.3cm}

\begin{itemize}
    \item \textbf{Feature Learning Network}: turns a raw $\sim$100K-point LiDAR cloud into a dense voxel-wise feature grid.
    \begin{enumerate}
        \item \textbf{Voxel Partition}: evenly partition 3D space into a $D\times H\times W$ voxel grid.
        \item \textbf{Grouping}: assign each point to its voxel; densities are highly uneven.
        \item \textbf{Random Sampling}: cap each voxel at $T$ points $\rightarrow$ (i) compute savings, (ii) reduces sampling bias from per-voxel point-count imbalance, adds training variation.
        \item \textbf{Stacked Voxel Feature Encoding (VFE)} -- per non-empty voxel $V = \{\mathbf{p}_i = [x_i, y_i, z_i, r_i]^\top\}_{i=1..t}$ ($r_i$ = LiDAR return intensity):
        \begin{itemize}
            \item Augment each point with its offset to voxel centroid: $\hat{\mathbf{p}}_i = [x_i, y_i, z_i, r_i, x_i-v_x, y_i-v_y, z_i-v_z]^\top \in \mathbb{R}^7$.
            \item Shared MLP (linear $+$ BN $+$ ReLU) $\rightarrow$ per-point features $\rightarrow$ max-pool across the voxel for a voxel-level feature $\rightarrow$ concat back per point (PointNet-seg local$+$global).
            \item Stack VFE layers; final = one feature per non-empty voxel ($\rightarrow$ sparse 4D tensor).
        \end{itemize}
    \end{enumerate}
    \item \textbf{Convolutional Middle Layers}: stack of \textit{3D conv $+$ BN $+$ ReLU} blocks on the sparse voxel tensor. Progressively expanding receptive field aggregates voxel-wise features into richer shape descriptions. \\ \textit{Key step}: collapse the vertical ($z$) dimension to a singleton $\rightarrow$ output is a 2D \textbf{BEV (Bird's-Eye-View) feature map}. Driving is essentially planar, so dropping height makes downstream 2D detection feasible.
    \item \textbf{Region Proposal Network}: standard 2D Faster-R-CNN-style RPN, now applicable because the middle layers already produced a BEV map. Two specializations for driving: (i) anchors only at \textit{two} orientations (longitudinal + frontoparallel) since vehicles in traffic appear in essentially two poses, (ii) regression outputs \textbf{7-DoF 3D boxes} $(x,y,z,w,l,h,\theta)$ instead of 4-DoF 2D ones -- 2D detection geometry, 3D output.
\end{itemize}

\subparagraph{Bounding Box Regression}
GT box parameterized as $(x_c^g, y_c^g, z_c^g, l^g, w^g, h^g, \theta^g)$ (center, length/width/height, yaw around $z$); anchor analogously with superscript $a$. Residual target $\mathbf{u}^* \in \mathbb{R}^7$ encoding the GT box \textit{relative} to the anchor:
$$\Delta x = \tfrac{x_c^g - x_c^a}{d^a},\ \Delta y = \tfrac{y_c^g - y_c^a}{d^a},\ \Delta z = \tfrac{z_c^g - z_c^a}{h^a}, \quad \Delta l = \log\tfrac{l^g}{l^a},\ \Delta w = \log\tfrac{w^g}{w^a},\ \Delta h = \log\tfrac{h^g}{h^a}, \quad \Delta\theta = \theta^g - \theta^a$$
with the anchor's BEV diagonal $d^a = \sqrt{(l^a)^2 + (w^a)^2}$. Normalizing BEV centers by $d^a$ (and height offset by $h^a$) makes the regression \textit{scale-invariant} in anchor size; \textit{log-ratios} for dimensions turn multiplicative size errors into additive ones and guarantee positivity after $\exp$. Direct analog of Faster R-CNN's 2D box parameterization, extended to 3D + rotation.

\subparagraph{Loss}
Multi-task loss, Faster-R-CNN style:
$$L = \alpha\, \tfrac{1}{N_{\text{pos}}}\sum_i L_{\text{cls}}(p_i^{\text{pos}}, 1) + \beta\, \tfrac{1}{N_{\text{neg}}}\sum_j L_{\text{cls}}(p_j^{\text{neg}}, 0) + \tfrac{1}{N_{\text{pos}}}\sum_i L_{\text{reg}}(\mathbf{u}_i, \mathbf{u}_i^*)$$
\begin{itemize}
    \item \textbf{Classification}: cross-entropy on both \textit{positive} and \textit{negative} anchors (separately normalized), weighted by $\alpha, \beta$ to balance their relative contribution.
    \item \textbf{Regression}: smooth-$L_1$ on the 7-D residual $\mathbf{u}_i$, \textit{gated to positive anchors only} -- negatives have no GT box to regress against (same gating as Faster R-CNN RPN).
\end{itemize}
\textit{Note:} Detection range and per-voxel cap $T$ are tuned \textit{per class}: cars use a wider range ($\sim 70$\,m) with $T=35$; pedestrians/cyclists a shorter range ($\sim 48$\,m) with $T=45$ -- smaller targets are denser per voxel but only relevant nearby.

\textit{Why operate in 3D?} Native 3D operations factor out pose variations and let the model directly exploit geometric structure (planarity, symmetry, repetition) that 2D projections obscure -- e.g.\ the VFE's centroid-relative offsets are a 3D-native prior.

\paragraph{PointRCNN}

\vspace{-0.3cm}
\begin{figure}[H]
    \centering
    \includegraphics[width=\linewidth]{img/PointRCNN.png}
    \caption{PointRCNN Architecture}
    \label{fig:PointRCNN}
\end{figure}
\vspace{-0.3cm}

\begin{itemize}
    \item \textbf{Bin-based 3D Box Generation}: per foreground point $p$, instead of regressing the box center as one continuous offset, \textit{discretize the location space into bins} and use a hybrid \textbf{classification + residual regression} head.
    \begin{itemize}
        \item Around $p$, a square location space of half-side $\mathcal{S}$ is split into $\mathcal{C}$ bins per dimension (bin width $\delta = 2\mathcal{S}/\mathcal{C}$). For $u\in\{x,y\}$: $\text{bin}_u^{(p)} = \lfloor (u^p - u^{(p)} + \mathcal{S})/\delta \rfloor$ (GT-center bin index), with residual $\text{res}_u^{(p)}$ measuring the offset within the bin center. Same for yaw $\theta$.
        \item Other dimensions ($z, h, w, l$) are regressed directly (no binning -- height range is small, sizes are smooth).
        \item \textbf{Hybrid loss}: cross-entropy on bin indices (coarse localization) + smooth-$L_1$ on residuals (fine offset), averaged over positive proposals. \textit{No objectness loss}: proposals come from foreground points, so they're always positive by construction.
        \item \textit{Why}: classification is more \textit{robust for coarse localization} than direct regression -- splitting into coarse-class + fine-residual stabilizes training, especially for far/sparse points.
    \end{itemize}
    \item \textbf{Canonical Transformation} (stage-2 refinement): for each proposal, transform its pooled points into a \textit{box-local} Canonical Coordinate System (CCS) -- origin at the proposal center, axes aligned with its orientation. Each proposal gets its own CCS.
    \begin{itemize}
        \item \textit{Why}: removes the proposal's global pose dependency $\rightarrow$ all boxes look alike in their own frame $\rightarrow$ refinement network learns local geometry consistently across proposals.
        \item \textit{What is transformed vs kept global}: only the raw spatial coordinates of pooled points are canonicalized; the \textit{semantic/high-level features} from stage 1 stay in the original frame (they encode global scene context).
        \item Canonical spatial features + global semantic features are then \textit{merged} (concat $\rightarrow$ MLP, then PointNet++ encoder), feeding two refinement heads: a second bin-based 3D box head + a confidence prediction.
    \end{itemize}
\end{itemize}


\subsection{3D Reconstruction and Localization}
\begin{definition}[Localization in CV.]
    CV is the inverse algorithmic process of inferring properties of the world and of optical sensors based on visual sensory observations. The localization them estimates the 3D Euclidean motion as a composite of translation $t$ and rotation $R$.
\end{definition}


\paragraph{Transformation conventions.} The forward (world/frame 1 $\to$ sensor/frame 2) and inverse maps are
\begin{equation*}
    \mathbf{P}_s = [\mathbf{R}\mid\mathbf{t}]\,\tilde{\mathbf{P}}_w \;\Leftrightarrow\; \mathbf{P}_s = \mathbf{R}\mathbf{P}_w + \mathbf{t} \;\Leftrightarrow\; \mathbf{P}_w = \mathbf{R}^T(\mathbf{P}_s - \mathbf{t}) \;\Leftrightarrow\; \mathbf{P}_w = \mathbf{R}^T\mathbf{P}_s - \mathbf{R}^T\mathbf{t}.
\end{equation*}
The translation vector is always \textit{the origin of the input frame expressed in coordinates of the output frame}. Hence: $\mathbf{R}, \mathbf{t}$ map world$\to$sensor; the inverse uses $\mathbf{R}^T$ (since $\mathbf{R}$ orthogonal $\Rightarrow \mathbf{R}^{-1}=\mathbf{R}^T$) and translation $-\mathbf{R}^T\mathbf{t} = \mathbf{C}$, the \textit{sensor origin (camera center) in world coordinates}.


\subsubsection{Motion Estimation from 3D and ICP Algorithm}
Motion estimation from \textbf{3D point clouds} aims to find the 3D motion $R,t$ of the sensor that best aligns two point clouds $\mathcal{P}_S, \mathcal{P}_{S'}$ obtained from the same static object (we get the movement relative to this point, ICP sensitive to such outliers --- use robust Loss rather than default squared Euclidean norm).\\
Since we do not know the point correspondences we must apply some approximation. 

\paragraph{Iterative Closest Point (ICP).}
ICP solves the correspondence problem as an optimization. Given two sets of 3D points $\mathcal{P}_s = [\mathbf{P}_s^{(1)}\;\mathbf{P}_s^{(2)}\;\dots\;\mathbf{P}_s^{(n)}]$ and $\mathcal{P}_{s'} = [\mathbf{P}_{s'}^{(1)}\;\mathbf{P}_{s'}^{(2)}\;\dots\;\mathbf{P}_{s'}^{(n)}]$, find the 3D Euclidean motion $\{\hat{\mathbf{R}}, \hat{\mathbf{t}}\}$ and the assignment function $\hat\phi: \{1,\dots,n\}\to\{1,\dots,n\}$ that minimize the sum of squared errors:
$$
\{\hat{\mathbf{R}}, \hat{\mathbf{t}}, \hat\phi\} = \arg\min_{\mathbf{R},\mathbf{t},\phi} E(\mathbf{R},\mathbf{t},\phi), \qquad E(\mathbf{R},\mathbf{t},\phi) = \sum_{i=1}^{n} \left\| \mathbf{R}\mathbf{P}_s^{(\phi(i))} + \mathbf{t} - \mathbf{P}_{s'}^{(i)} \right\|_2^2,
$$
where $\mathbf{P}_s^{(\phi(i))}$ and $\mathbf{P}_{s'}^{(i)}$ are the corresponding points induced by the assignment function $\phi$. The two clouds need \textit{not} be equal-sized: $\phi$ is a nearest-neighbour (many-to-one) map, not a bijection, so $n_s \neq n_{s'}$ is fine.

\begin{itemize}
    \item \textbf{Key idea}: if the correct correspondences $\phi$ are \textit{known}, the optimal $\{\hat{\mathbf{R}}, \hat{\mathbf{t}}\}$ has a \textbf{closed-form} solution.
    \item \textbf{Intuition}: at the optimal motion, corresponding points are closer to each other than to any other point.
    \item \textbf{Heuristic / caveat}: $\phi$ matches each point to its \textit{currently} closest point $\Rightarrow$ a \textbf{good initialization} is required (converges to local minimum only).
\end{itemize}
\textbf{Procedure} (alternate assignment $\leftrightarrow$ motion until convergence):
\begin{enumerate}
    \item Initial guess $\{\check{\mathbf{R}}, \check{\mathbf{t}}\}$ (usually $I$ and $0$).
    \item \textit{Assignment}: associate each point in $\mathcal{P}_{s'}$ with the closest point in $\mathcal{P}_s$ (defines $\phi$; need not be bijective $\Rightarrow$ no perfect matching of $\mathcal{P}_{s'}$ \& $\mathcal{P}_s$ required - duplicate assignments in $\mathcal{P}_s$ possible).
    \item \textit{Motion}: solve the closed-form optimal $\{\hat{\mathbf{R}}, \hat{\mathbf{t}}\}$ for the fixed $\phi$.
    \item Repeat until convergence.
\end{enumerate}

\paragraph{Optimal Motion.}
Step (3) of finding the optimal motion $\{\hat{\mathbf{R}}, \hat{\mathbf{t}}\}$ given $\phi$ from Step (2) solved using \textbf{Least Squares Linear Regression}.\\

\textbf{Caveat (}$\mathbf{R}\notin$\textbf{ vector space):} translation $\mathbf{t}$ is plain least squares, but $\mathbf{R}\in SO(3)$ (special orthogonal group) has $9$ params with only $3$ DOF (entries interdependent) $\Rightarrow$ not amenable to default least squares. Strategy: \textit{eliminate} $\mathbf{t}$ symbolically, reduce to a rotation-only problem, solve by SVD.\\

\textbf{1. Optimal translation given rotation.} Fix $\hat{\mathbf{R}}$. Each pair can be aligned exactly ($\hat{\mathbf{R}}\mathbf{P}_s^{(\phi(i))} + \mathbf{t} - \mathbf{P}_{s'}^{(i)} = 0$); stacking all $n$ gives an overdetermined $3n\times 3$ system $\mathbf{A}\mathbf{t}=\mathbf{b}$ with $\mathbf{A}=[\mathbb{I}_3;\dots;\mathbb{I}_3]$, $\mathbf{b}^{(i)}=\mathbf{P}_{s'}^{(i)}-\hat{\mathbf{R}}\mathbf{P}_s^{(\phi(i))}$. Normal equations $\mathbf{t}=(\mathbf{A}^T\mathbf{A})^{-1}\mathbf{A}^T\mathbf{b}$ with $\mathbf{A}^T\mathbf{A}=n\mathbb{I}_3$ yield (a formula in $\mathbf{R}$, not yet a number):
$$
\hat{\mathbf{t}} = \boldsymbol{\mu}_{s'} - \hat{\mathbf{R}}\,\boldsymbol{\mu}_{s,\phi}.
$$
\textit{Intuition}: the centroids of the two \textit{matched} sets are aligned by $\{\hat{\mathbf{R}},\hat{\mathbf{t}}\}$. Note $\boldsymbol{\mu}_{s,\phi}\neq\boldsymbol{\mu}_s$ in general (only $\phi$-selected points contribute, $\phi$ not bijective).

\textbf{2. Optimal rotation.} Substitute $\hat{\mathbf{t}}(\mathbf{R})$ back into $E$ $\Rightarrow$ $\mathbf{t}$ cancels, objective is centered and depends on $\mathbf{R}$ only. Minimizing it $\Leftrightarrow$ maximizing
$$
\max_{\mathbf{R}}\sum_{i=1}^{n}\big(\mathbf{P}_s^{(\phi(i))}-\boldsymbol{\mu}_{s,\phi}\big)^T\mathbf{R}^T\big(\mathbf{P}_{s'}^{(i)}-\boldsymbol{\mu}_{s'}\big).
$$
Writing $\mathbf{R}^T=\mathbf{U}\mathbf{V}^T$ (composition of two independent rotations) reduces this to $\max_{\mathbf{U},\mathbf{V}}\operatorname{trace}(\mathbf{U}^T\mathbf{W}\mathbf{V})$, where
$$
\mathbf{W}=\sum_{i=1}^{n}\big(\mathbf{P}_s^{(\phi(i))}-\boldsymbol{\mu}_{s,\phi}\big)\big(\mathbf{P}_{s'}^{(i)}-\boldsymbol{\mu}_{s'}\big)^T
$$
is the \textbf{cross-covariance} of the two centered point sets.

\textbf{3. Solve via SVD.} The trace-maximization is solved exactly by the SVD $\mathbf{W}=\hat{\mathbf{U}}\boldsymbol{\Sigma}\hat{\mathbf{V}}^T$. To stay in $SO(3)$ (reject reflections $\to$ non-right-handed frames, which lie in $O(3)\setminus SO(3)$), correct with the determinants:
$$
\hat{\mathbf{R}} = \hat{\mathbf{V}}\,\operatorname{diag}\!\big(1,\,1,\,\det(\hat{\mathbf{U}})\det(\hat{\mathbf{V}})\big)\,\hat{\mathbf{U}}^T.
$$
\textit{Intuition}: align the principal directions of variation (modeled by the cross-covariance) of the two corresponding sets.

\textbf{Order note}: the derivation eliminates $\mathbf{t}$ first to expose the rotation-only objective; at \textit{runtime} compute $\hat{\mathbf{R}}$ (SVD) first, then $\hat{\mathbf{t}}=\boldsymbol{\mu}_{s'}-\hat{\mathbf{R}}\boldsymbol{\mu}_{s,\phi}$.

\subsubsection{Motion Estimation from 2D and VO}
We no longer leverage 3D Point Clouds to estimate motion but rather 2D images from various viewpoints. 

\begin{definition}[Visual Odometry.]
    Assuming a single monocular system with images taken at time $k: I_{0:n}=\{I_0,\dots,I_n\}$, then two sequential images at $k-1$ and $k$ are relate by the rigid body transformation:
    $$
    T_k = 
    \begin{bmatrix}
        R_{k,k-1} & t_{k,k-1}\\
        0 & 1
    \end{bmatrix}
    $$
    Trough the concatenation of all transformations $T_{0:n}$ we can recover the full trajectory at each camera position $C_1, \dots ,C_n$ relative to the initial coordinate frame $C_0$ (pose by pose/ 2D to 2D).
\end{definition}

This works well given \textbf{sufficient illumination}, a \textbf{dominance of static} over dynamic scene constituents, \textbf{adequate texture} for informative feature extraction, and \textbf{sufficient co-visibility} between at least consecutive frame pairs.


\subsubsection{Epipolar Geometry, Two-frame SFM, and the 8-Point Algorithm}
\paragraph{Epipolar Geometry and Constraint}
For a point $p$ in $k-1$ the matching point in $k$ must lie on the \textit{epipolar line} $l'$ in the second image (\textbf{Epipolar Constraint}), where 
\begin{itemize}
    \item \textbf{epipolar line} $l'$: projection of the back-projected ray $\pi^{-1}(p) = \lambda K^{-1}p$ onto the camera image
    \item \textbf{epipole}: the projection of the optical center on the camera image; i.e., where the epipolar lines intersect (can be outside the image).
    \item \textbf{epipolar plane} is given by the hyperplane via the camera centers $C_r,C_l$ and the image point
\end{itemize}
\vspace{-10px}

\begin{figure}[H]
    \centering
    \includegraphics[width=0.6\linewidth]{img/EpipolarConstraint.png}
    \caption{The Epipolar Constraint of Converging Cameras}
    \label{fig:epipolarConstraint}
\end{figure}

\begin{itemize}
    \item Converging cameras: As the position of the 3D point changes the epipolar lines rotate around the baseline
    \item Identical and horizontally-aligned cameras (Simplified Case): $e$'s are at infinity, since the epipolar lines are essentially parallel - so the cameras never see each other no matter the size of $b$.
    \item Sequential images in forward motion: Epipolar with constant coordinates in both images (imaging the epipolar lines intersecting in the middle of the image at a focus point). The points move along these lines.
\end{itemize}


\paragraph{Essential Matrix} 
Given two normalized (homogeneous) image points
$\bar{\mathbf p}_1,\bar{\mathbf p}_2$ and the relative pose $(R,\mathbf T)$, we derive the essential matrix ($3\times 3$; 5DOF: 3 by $R$, 2 by $t$ - translation up to scale only) from coplanarity of $\bar{\mathbf p}_2$, $\bar{\mathbf p}'_1$, and $\mathbf T$:
\[
\begin{aligned}
\bar{\mathbf p}'_1 &= R\,\bar{\mathbf p}_1,\\
\mathbf n &= \mathbf T \times \bar{\mathbf p}'_1 \qquad \text{(normal of the epipolar plane)},\\
\bar{\mathbf p}_2^\top \mathbf n &= 0\\
\Rightarrow\ \bar{\mathbf p}_2^\top\big(\mathbf T \times (R\bar{\mathbf p}_1)\big) &= 0\\
\Rightarrow\ \bar{\mathbf p}_2^\top[\mathbf T]_\times R\,\bar{\mathbf p}_1 &= 0\\
\Rightarrow\ \bar{\mathbf p}_2^\top E\,\bar{\mathbf p}_1 &= 0,
\qquad
E = [\mathbf T]_\times R .
\end{aligned}
\]
\noindent\textit{Intuition:} $[\mathbf T]_\times$ turns “cross with $\mathbf T$” into a matrix, so $E$
packages translation (up to scale) and rotation into one bilinear constraint that corresponding
\emph{normalized} points must satisfy.

\paragraph{8-Point Algorithm for the Essential Matrix.} Since each pair of \textit{normalized} ($\mathbf{\tilde{p}}_1^{(i)}=\mathbf{K}^{-1}\mathbf{p}_1^{(i)}$ and zero-centered with std $\sqrt{2} \rightarrow  \mathbf{\bar{p}}_1^{(i)}$) point correspondences provides a linear equation through the Epipolar constraint $\bar{\mathbf p}_2^\top E\,\bar{\mathbf p}_1 = 0$. Normalization is required since we have elementwise multiplication over columns that have different scales in orders of magnitude, that would lead to numerical instability:
$$
\bar{u}_2 \bar{u}_1 e_{11}+\bar{u}_2 \bar{v}_1 e_{12}+\bar{u}_2 e_{13}+\bar{v}_2 \bar{u}_1 e_{21}+\bar{v}_2 \bar{v}_1 e_{22}+\bar{v}_2 e_{23}+\bar{u}_1 e_{31}+\bar{v}_1 e_{32}+e_{33}=0
$$

We can stack these equations for $n$ correspondences into a single homogeneous linear system
\[
Q\,\bar{\mathbf E} = \mathbf 0,
\qquad
\bar{\mathbf E}=\mathrm{vec}(E)=
\begin{bmatrix}
e_{11}&e_{12}&e_{13}&e_{21}&e_{22}&e_{23}&e_{31}&e_{32}&e_{33}
\end{bmatrix}^{\!\top},
\]
where each row of $Q\in\mathbb{R}^{n\times 9}$ is built from one correspondence
$(\bar u_1,\bar v_1)\leftrightarrow(\bar u_2,\bar v_2)$:
\[
Q_i=
\begin{bmatrix}
\bar u_2\bar u_1 & \bar u_2\bar v_1 & \bar u_2 &
\bar v_2\bar u_1 & \bar v_2\bar v_1 & \bar v_2 &
\bar u_1 & \bar v_1 & 1
\end{bmatrix}.
\]

\textbf{Minimal case ($n=8$).}
Each correspondence contributes one independent linear constraint, so $8$ points are sufficient to determine
$\bar{\mathbf E}$ up to scale (in practice $Q$ should have rank $8$ so that $\mathrm{null}(Q)$ is 1D).\\

\textbf{Over-determined case ($n>8$).}
Solve in least-squares form
\[
\min_{\|\bar{\mathbf E}\|=1}\ \|Q\bar{\mathbf E}\|^2,
\]
whose solution is the right singular vector of $Q$ associated with the smallest singular value
(equivalently, the eigenvector of $Q^\top Q$ with smallest eigenvalue). Reshape $\bar{\mathbf E}$ into
$E\in\mathbb{R}^{3\times 3}$.\\

\textbf{Enforcing the essential-matrix structure (rank-2 constraint).}
Because $E=[\mathbf T]_\times R$, a true essential matrix has \emph{two equal non-zero singular values and one zero} (so $\mathrm{rank}(E)=2$, $5$ DOF). The linear estimate $\hat E$ from noisy data generally violates this, so we project it onto the closest valid essential matrix with a \textit{second} (now only $3\times3$) SVD, averaging the two largest singular values and forcing the smallest to zero:
\[
\hat E = U_E\,\Sigma_E\,V_E^\top
\quad\longrightarrow\quad
E^\ast = U_E\,
\mathrm{diag}\!\Big(\tfrac{\sigma_E^{(1)}+\sigma_E^{(2)}}{2},\ \tfrac{\sigma_E^{(1)}+\sigma_E^{(2)}}{2},\ 0\Big)\,
V_E^\top .
\]

\textbf{Undoing the normalizations.}
Finally reverse the pre-processing in the opposite order: since $\bar{\mathbf p}=N\,\tilde{\mathbf p}$ turns the constraint into $\tilde{\mathbf p}_2^\top(N_2^\top E^\ast N_1)\tilde{\mathbf p}_1=0$, apply the inverse linear transforms of the std-$\sqrt2$ conditioning (step~2) to map $E^\ast$ back to the original normalized-image coordinates.\\

\textbf{Degenerate configurations.}
The linear 8-point estimation becomes ill-conditioned/degenerate when the underlying 3D points are (nearly)
coplanar; in such cases alternative minimal solvers (e.g.\ 5-point methods) are preferred.

\paragraph{Recovering $(R,\mathbf T)$ — the 4 Solutions.} A final $3\times3$ SVD $E^\ast=U_E\,\Sigma_{E^\ast}V_E^\top$ decomposes the essential matrix into motion. Because the sign of the translation and the direction of rotation interfere, this yields \textbf{four} candidate motions, built from the third column $\mathbf u_3$ of $U_E$ and the fixed orthogonal matrix
\vspace{-0.3cm}
\[
W=\begin{bmatrix}0&-1&0\\ 1&0&0\\ 0&0&1\end{bmatrix}:
\quad
\mathbf T=\pm\mathbf u_3,
\qquad
R=U_E W V_E^\top \ \text{ or }\ U_E W^\top V_E^\top .
\]
Any candidate $R$ with $\det R=-1$ is negated to make it a proper rotation ($\det R=+1$). The single physically valid motion is selected by \textbf{triangulating} the correspondences and keeping the solution for which the reconstructed 3D points lie \emph{in front of both cameras} ($z>0$, the cheirality check). This \textbf{gives us $\mathbf T$ up to scale}.

\paragraph{2D-to-2D VO Pipeline.} One iteration of the monocular VO loop: (1) capture $I_k$; (2) match features $I_{k-1}\!\leftrightarrow\! I_k$; (3) estimate $E$ (8-point); (4) decompose into $R_k,\mathbf t_k$, form $T_k$; (5) compute relative scale and rescale $\mathbf t_k$; (6) concatenate the global pose $C_k = C_{k-1}T_k$; repeat.

\paragraph{Relative Scale.} The absolute scale is unrecoverable from images, but the \textit{relative} scale between consecutive transforms is enough for a consistent trajectory. Triangulate the same points from two subsequent pairs ($X_{k-1},X_k$); the scale is the distance ratio over co-observed points $i,j$ (averaged/RANSAC'd in practice):
\[
r = \frac{\lVert X_{k-1,i}-X_{k-1,j}\rVert}{\lVert X_{k,i}-X_{k,j}\rVert}.
\]
\vspace{-0.8cm}
\subsubsection{RANSAC}
Sensor noise, image blur, algorithmic noise (like greedy algos) etc, we must do outlier rejection via \textit{robost estimation}. \textbf{RANSAC} is the standard approach for fitting a model in the presence of many outliers (e.g. $E$).

\begin{itemize}
    \item \textbf{Model in SfM:} $E$ (calibrated) (or $F$ if uncalibrated); equivalently $(R,T)$ once calibrated.
    \item \textbf{Minimal sample size:} depends on the solver (e.g. $m{=}8$ for the classic 8-point $E$).
    \item \textbf{Distance to the model:} choose a residual $r_i(\theta)$, e.g.\ some geometric or (symmetric) epipolar line distance
\end{itemize}

\begin{itemize}
    \item \textbf{Idea:} fit many hypotheses from random minimal samples and keep the one with the largest inlier set.
    \item \textbf{Loop:} sample $m$ points $\rightarrow$ fit $\theta$ $\rightarrow$ compute residuals $\rightarrow$ inliers if $r_i(\theta)<\tau$.
    \item \textbf{Output:} best hypothesis + \textbf{refit} using all inliers (least squares).
    \item \textbf{Notes:} non-deterministic; robust to bad initialization; quality depends on $\tau$, $m$, and $k$.
\end{itemize}

\paragraph{Required Iterations.} For inlier fraction $w$, sample size $m$, and success probability $p$, the iterations $k=\frac{\log(1-p)}{\log(1-w^m)}$ grow fast with the outlier rate and $m$ (e.g.\ $w=50\%,p=99\%$: $m=8\!\to\!k\approx1177$ vs.\ $m=2\!\to\!k\approx16$). Hence smaller minimal samples via motion/scene constraints (planar $\to$ 2-point, vehicle $\to$ 1-point) drastically cut cost.

\subsubsection{Learned Feature Mapping and SuperGlue}
\textbf{Approaches:}
\begin{itemize}
    \item \textbf{Nearest-neighbor matching:} compare individual descriptors and link closest pairs — greedy, locally optimal, ignores assignment structure.
    \item \textbf{Learned matching (SuperGlue):} cast both objectives as one \textit{differentiable global optimization} over all features in both frames, solved end-to-end rather than per-keypoint.
\end{itemize}

\paragraph{SuperGlue: Differentiable Optimal Matching.} Since each keypoint matches at most one (or none), the matching is a \textbf{partial assignment} between the keypoint sets $\mathcal A=\{1,\dots,M\}$ and $\mathcal B=\{1,\dots,N\}$. SuperGlue makes this \textit{soft and differentiable}: every possible pair $(i,j)$ gets a \textbf{confidence} $p_{ij}\in[0,1]$ computed from the descriptors $\mathbf f_i^{(1)},\mathbf f_j^{(2)}$ ($p_{ij}=1$ = certain match). The partial-assignment property is enforced by the constraints
\[
\sum_{j=1}^{N} p_{ij}\le 1\ \ \forall i\in\mathcal A,
\qquad
\sum_{i=1}^{M} p_{ij}\le 1\ \ \forall j\in\mathcal B,
\]
i.e.\ each keypoint distributes at most unit confidence (the slack $<1$ absorbs unmatched keypoints).


\begin{figure}[H]
    \centering
    \includegraphics[width=\linewidth]{img/SuperGlueDifferentiableOptimalmatching.png}
    \caption{SuperGlue Differentiable Optimal Matching}
    \label{fig:SuperGlueDifferentiableOptimalmatching}
\end{figure}
\textbf{Pipeline} (Fig.~\ref{fig:SuperGlueDifferentiableOptimalmatching}):

\begin{enumerate}
    \item \textbf{Score matrix.} $\mathbf S_{i,j}=\mathbf f_i^{(1)}\!\cdot\mathbf f_j^{(2)}$ ($M\times N$): dot-product similarity, more similar descriptors $\Rightarrow$ larger score. $\rightarrow$ raw, unnormalized affinities, not yet valid confidences.
    \item \textbf{Dustbin augmentation.} Append one row \& column with a \textit{learnable shared} score $z$, giving $\bar{\mathbf S}$ of size $(M{+}1)\times(N{+}1)$. The dustbin soaks up unmatched keypoints, turning the $\le 1$ constraints into equalities (each row/column, dustbin included, sums to 1 $\Rightarrow$ normalization). $\rightarrow$ a ``trash can'' that lets a row-sum still reach 1 when a keypoint has no real partner; this is what allows non-matching.
    \item \textbf{Sinkhorn.} Exponentiate $\bar{\mathbf S}$, then iterate alternating row- and column-normalization for $T$ steps. This converges to a doubly-stochastic soft assignment $\bar{\mathbf P}$ that satisfies the constraints differentiably. $\rightarrow$ fixing rows breaks columns and vice versa, so alternating converges to satisfy both; a soft, gradient-friendly stand-in for the hard Hungarian assignment.
    \item \textbf{Output.} $\bar{\mathbf P}:(M{+}1)\times(N{+}1)$ partial assignment; drop the dustbin row/column to read off matches. $\rightarrow$ confidence left in the dustbin means ``no match''.
    \item \textbf{Training.} Maximum-likelihood loss $L=-\sum_{(i,j)\in\mathcal M}\log \mathbf P_{ij}$ over GT (non-)matches $\mathcal M$, back-propagated into the learnable descriptors. $\rightarrow$ rewards high probability on true matches, shaping descriptors so correct pairs score highest.
\end{enumerate}

\textbf{Input: Attentional Context Aggregation} (how do we get the context-aware descriptors $\mathbf f_i^{(1)},\mathbf f_j^{(2)}$ fed to the optimal matching). Raw local descriptors (e.g.\ SIFT) are purely \textit{local}; matching needs descriptors \textbf{aware of the global context} (spatial layout in both frames). Achieved with attention, treating each keypoint as a \textbf{token}:
\begin{figure}[H]
    \centering
    \includegraphics[width=0.7\linewidth]{img/SuperGlueFeatures.png}
    \caption{SuperGlue Attentional Context Aggregation}
    \label{fig:SuperGlueFeatures}
\end{figure}
\begin{enumerate}
    \item \textbf{Keypoint Encoder.} MLP embeds position \& confidence $(x_i,y_i,c_i)$ and \textit{adds} it to the visual descriptor $\mathbf d_i$, giving the initial embedding $\mathbf x_i^{(0)}$. $\rightarrow$ fuses \textit{appearance + location} into one token.
    \item \textbf{Alternating attention ($L$ layers).} Residual attention block $\mathbf X^{(l)}=\mathbf X^{(l-1)}+\mathrm{MLP}\big(\mathrm{softmax}(\mathbf Q^\top\mathbf K)\,\mathbf V\big)$, with $\mathbf Q,\mathbf K,\mathbf V$ learned projections of the embeddings:
    \begin{itemize}
        \item \textbf{Self-attention} (odd layers): $\mathbf K,\mathbf V$ from the \textit{same} image $\rightarrow$ attend to adjacent/salient/similar-looking keypoints (intra-frame layout).
        \item \textbf{Cross-attention} (even layers): $\mathbf K,\mathbf V$ from the \textit{other} image $\rightarrow$ attend to keypoints near the actual match (inter-frame comparison).
    \end{itemize}
    $\rightarrow$ cascading both makes each keypoint aware how its neighbors and salient keypoints map to the other image.
    \item \textbf{Final projection.} Linear layer $\mathbf f_i^{(1)}=\mathbf W\,\mathbf x_i^{(2L)}$ yields the context-aware descriptors fed to the matching stage.
\end{enumerate}


\subsubsection{Neural Radiance Fields (NeRF)}
\textbf{Why:} Classical 3D reps each lose something: point clouds / meshes / voxels are \textbf{discrete} (info loss); multi-view images are only 2D projections; implicit functions need GT geometry. \textbf{NeRF:} store the scene as \textit{one continuous function} (the MLP weights), learned from \textit{posed images alone} (no GT geometry). $\rightarrow$ effectively infinite resolution from a few MB of weights; goal is \textbf{novel-view synthesis} (render the scene from unseen camera poses).\\

\textbf{Core idea:} $F_\Theta:\ \underbrace{(x,y,z)}_{\text{position}}+\underbrace{(\theta,\phi)}_{\text{view dir.}}\ \mapsto\ \underbrace{\sigma}_{\text{density}}+\underbrace{(r,g,b)}_{\text{radiance}}$. Density $=$ \textit{view-independent geometry} (does a surface exist here?); radiance $=$ \textit{view-dependent color} (a glossy highlight shifts as you move, the surface does not, hence the split). Render via volume rendering $\rightarrow$ fully differentiable $\rightarrow$ train on a pixel reconstruction loss.
\vspace{-0.5cm}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.8\linewidth]{img/NeRF.png}
    \caption{Neural Radiance Fields}
    \label{fig:NeRF}
\end{figure}

\textbf{Architecture trick} (enforces view-independent $\sigma$): MLP$_1(\mathbf x)\to\sigma+\text{feature }\mathbf f$; then append $(\theta,\phi)$ to $\mathbf f$ $\to$ MLP$_2\to\mathbf c$. So geometry cannot change with viewpoint by construction; only color can. $\rightarrow$ prevents the net from ``cheating'' by faking geometry per view, which would break multi-view consistency.\\

\textbf{Volume rendering} (how the MLP becomes a pixel): read $\sigma(\mathbf x)$ as the \textbf{probability density of ray termination} at $\mathbf x$ (likelihood a photon is absorbed/reflected there). A pixel is the expected radiance along its ray $\mathbf r(t)=\mathbf o+t\mathbf d$:
\[
\mathbf C(\mathbf r)=\int_{t_n}^{t_f} \underbrace{T(t)}_{\text{reach }t}\,\underbrace{\sigma}_{\text{stop here}}\,\mathbf c\,dt,
\qquad
T(t)=\exp\!\Big(-\!\int_{t_n}^{t}\sigma\,ds\Big)\ \text{(transmittance)}.
\]
\textbf{Intuition:} $T$ $=$ chance the ray survived this far; at a surface $\sigma$ spikes $\Rightarrow$ $T$ collapses $\Rightarrow$ the first opaque hit dominates the integral $\Rightarrow$ occlusion is handled for free, and empty space ($\sigma\!\approx\!0$) simply contributes nothing.\\

\textbf{Discretize \& train:} $N$ stratified (jittered) samples per ray turn the integral into a weighted sum, segment opacity $\alpha_i=1-e^{-\sigma_i\delta_i}$:
\[
\hat{\mathbf C}(\mathbf r)=\sum_{i=1}^{N} T_i\,\alpha_i\,\mathbf c_i,
\quad
T_i=\textstyle\prod_{j<i}(1-\alpha_j),
\quad
\delta_i=t_{i+1}-t_i.
\]
This is just \textit{alpha compositing} of the samples front-to-back, and it is differentiable in $(\sigma_i,\mathbf c_i)$ $\Rightarrow$ the pixel loss back-props through rendering into $F_\Theta$ (no 3D supervision needed). \textit{Jittering} the samples avoids the net overfitting to fixed depths.\\

\textbf{Positional encoding} (why detail isn't blurry): MLPs are \textit{spectrally biased} to low frequencies, so raw $xyz\theta\phi$ inputs $\Rightarrow$ smooth/blurry output. Map each coord through fixed sinusoids of exponentially growing frequency, $F_\Theta=F'_\Theta\circ\gamma$:
\[
\gamma(p)=\big(\sin(2^0\pi p),\cos(2^0\pi p),\dots,\sin(2^{L-1}\pi p),\cos(2^{L-1}\pi p)\big),
\quad L{=}10\ (\mathbf x),\ L{=}4\ (\mathbf d).
\]
\textbf{Intuition:} nearby coords get very different encodings at high frequencies, handing the network a high-frequency basis (Fourier features) $\Rightarrow$ it can represent sharp edges \& fine texture instead of averaging them away.

\paragraph{Limitations} Basic NeRF is \textbf{per-scene} (one trained network $=$ one scene, no generalization), covers only a \textbf{bounded volume}, and assumes a \textbf{static scene}: fixed illumination/medium density, no transient or dynamic objects. $\rightarrow$ exactly the assumptions a \textit{driving} scene violates (photos span hours/seasons, cars \& pedestrians move, cities are huge), motivating the two extensions below.

\paragraph{Extensions for Automated Driving} The driving setting breaks NeRF's static + bounded assumptions, so two ideas adapt it:
\begin{itemize}
    \item \textbf{NeRF in the Wild} (handles \textit{appearance \& transient variation}): splits the scene into a \textbf{static} branch and a \textbf{transient} branch. Geometry (density) stays shared, but color is conditioned on a per-image \textbf{appearance embedding} (absorbs changing illumination/weather across photos) and a separate transient head with an \textbf{uncertainty} $\beta$ models dynamic objects so they don't corrupt the static reconstruction. $\rightarrow$ lets NeRF train on uncontrolled, time-varying captures instead of one clean static scene.
    \item \textbf{Block-NeRF} (handles \textit{scale}): a single MLP can't fit a city, so the area is tiled into \textbf{many overlapping Block-NeRFs}, one per intersection. Dynamic objects are \textbf{masked out} (semantic segmentation) during training, and each network also predicts \textbf{visibility} (transmittance of a ray). At render time, neighboring blocks are \textbf{composited} by inverse-distance weighting (low-visibility blocks discarded) $\rightarrow$ smooth transitions and \textbf{city-scale}, reconstruction from real driving logs.
\end{itemize}
\textit{Takeaway:} together they remove the static-scene and bounded-volume limitations $\Rightarrow$ NeRF becomes usable for large, dynamic, real-world driving environments (simulation, mapping, scene reconstruction).

\subsection{Domain Adaptation}
\subsubsection{Motivation, Taxonomy, and Fundamentals}
For SAE level 5 (full automation) we require \textit{unlimited} Operation Design Domain, by definition --- in normal and adverse conditions (generalization is often poor with domain shift).

\paragraph{Domain Shift Formal Definition} A \textbf{domain} is the joint distribution $p(I,Y)$ of inputs $I$ and reference outputs $Y$. Source samples $(I_s,Y_s)\sim p_s$, target samples $(I_t,Y_t)\sim p_t$. \textbf{Domain shift:} in general $p_s\neq p_t$ for (unseen) target data $\rightarrow$ models trained on $p_s$ generalize poorly to $p_t$.\\

\textbf{Domain Shifts} Categorized by what changes between source and target domain:
\begin{itemize}
    \item \textbf{Low-level} (appearance only like color, semantics unchanged): \textit{condition-level} shift (weather, day/night) and \textit{sensor} differences (camera model/optics).
    \item \textbf{High-level} (semantics/structure): \textit{semantic} shift in object size, shape, or spatial configuration, e.g.\ by geographic location.
    \item \textbf{Both:} \textit{synthetic vs.\ real} (rendering) combines a low-level (look) and high-level (layout) shift.
\end{itemize}

\paragraph{Why not just train on everything?} Naïve: add all ``outsider'' domains to labeled training set and large enough foundation model $\rightarrow$ basically impossible.
\begin{itemize}
    \item \textbf{Curse of dimensionality}: must cover \textit{all combinations} of shift factors (weather $\times$ sensor $\times$ location $\times\dots$).
    \item \textbf{Open domain}: test data can still be unseen/rare/unpredicted domain $\rightarrow$ unlimited ODD must handle it.
    \item \textbf{Annotation cost}: exhaustive labels for massive data are expensive, especially for dense tasks (segmentation).
    \item \textbf{Acquisition}: collecting data for narrow domains is hard: geographies, adverse conditions, dangerous collection.
\end{itemize}

\paragraph{Task Variants} Distinguished by which assets are available and \textit{when} (here: semantic segmentation):
\begin{itemize}
    \item \textbf{Unsupervised DA (UDA):} labeled source + \textit{unlabeled} target, at train time.
    \item \textbf{Weakly-supervised DA:} as UDA, but target has \textbf{correspondences} to source (e.g.\ same place day/night).
    \item \textbf{Source-free / model adaptation:} only a \textit{pretrained source model} (no source data) + unlabeled target.
    \item \textbf{Test-time / online DA:} pretrained model adapted on the fly to \textbf{unlabeled sequential test data}.
    \item \textbf{Domain generalization:} (large) labeled source(s) only; target \textbf{unseen}, \textit{no} adaptation at test time.
\end{itemize}

\subsubsection{Input-Level Adaptation}
Align the \textit{image distributions of the source and the target domain}: for \textit{low-level} domain shifts (e.g. fog/no fog).

\paragraph{Physics-based Domain Translation}
Leverage known priors of the domain shift: no target needed, reliable, controllable, and interpretable; but hard or impossible: assumptions necessary.\\

\textbf{Image-Based}: e.g fog/rain simulation via depth plane fitting and guided filters using (like NeRF) $\hat{t}(x)=e^{-\alpha l (x)}$ with attenuation coefficient $\alpha$ to control fog density.\\
This allows us to then train with the labels from the source domain (if available), removing the need for real target-domain data (pseudo-labels). Or use a segmentation model on unlabled source data, and apply these results (pseudo ground truth) as labels to the syntetic target data.\\

\textbf{Lidar-Based}: e.g.\ snowfall has particles, so no longer just attenuation (as in the clear-air lidar introduction) but also \textit{backscattering}. Idea: simulate small spheres in the medium as snowflakes, per lidar pulse (small opening angle, azimuth $\approx 0.1^\circ$). Each simulated particle $j$ intersects the beam with a distribution-driven diameter $D_j$, range $R_j$, and center; from these its \textbf{occlusion angle} $\theta_j$ follows, partitioning the beam as $\theta = \theta_0 + \sum^n_{j=1}\theta_j$ ($\theta_0$: un-occluded part reaching the target).
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\linewidth]{img/LidarSnowfall.png}
    \caption{Pulsed Lidar in Snowfall}
    \label{fig:LidarSnowfall}
\end{figure}

The \textit{target} response must also account for the snowflakes: since a flake is smaller than the pulse extent ($D_j\ll c\tau_H$), its impulse response is modeled as a Dirac delta. A flake's peak can exceed the real object's $\rightarrow$ snow-caused noisy (spurious) lidar points.\\

\textbf{Snowfall --- target impulse response.} The linear system model and optical channel $H_C(R)=\frac{T^2(R)}{R^2}\xi(R)$ are \textit{unchanged} (same as clear weather); only the \textit{target} response $H_T$ changes. The beam (opening angle $\theta$) is partly occluded by $n$ flakes, each blocking a sub-angle $\theta_j$ at range $R_j$, so $\theta=\theta_0+\sum_{j=1}^n\theta_j$ ($\theta_0$: fraction still reaching the real target). With $D_j\ll c\tau_H$ each flake is a Dirac in range:
\begin{equation*}
H_T(R)=\frac{1}{\theta}\Big(\underbrace{\rho_0\,\theta_0\,\delta(R-R_0)}_{\text{real target}}+\rho_s\sum_{j=1}^n\underbrace{\theta_j\,\delta(R-R_j)}_{\text{flake }j}\Big)=H_T^0(R)+\sum_{j=1}^n H_T^j(R),
\end{equation*}
i.e.\ a superposition weighted by each return's occlusion fraction $\theta_j/\theta$ and reflectivity ($\rho_s$ snow, $\rho_0$ target).\\

\textbf{Snowfall --- received waveform.} By linearity of the convolution the received power is the same superposition,
\begin{equation*}
P_{R,\text{snow}}(R)=P_{R,\text{snow}}^0(R)+\sum_{j=1}^n P_{R,\text{snow}}^j(R),
\end{equation*}
and substituting $H_T^j$ into the linear system model shifts/scales the emitted $\sin^2$ pulse to range $R_j$:
\begin{equation*}
\begin{aligned}
P_{R,\text{snow}}^j(R)&=\frac{C_A P_0\,\rho_s\,\theta_j\,\xi(R_j)}{\theta\,R_j^2}\int_0^{2\tau_H}\!\sin^2\!\Big(\tfrac{\pi}{2\tau_H}t\Big)\delta\!\Big(R-\tfrac{ct}{2}-R_j\Big)dt\\[1mm]
&=\begin{cases}\dfrac{C_A P_0\,\rho_s\,\theta_j\,\xi(R_j)}{\theta\,R_j^2}\sin^2\!\Big(\dfrac{\pi(R-R_j)}{c\tau_H}\Big), & R_j\le R\le R_j+c\tau_H,\\[2mm] 0,&\text{otherwise.}\end{cases}
\end{aligned}
\end{equation*}
and analogously $P_{R,\text{snow}}^0$ for the real target at $R_0$ (with $\rho_0,\theta_0$).\\

\textbf{Intuition:} each flake reflects back the beam fraction $\theta_j/\theta$ it occludes $\rightarrow$ one $\sin^2$ peak per flake at its range $R_j$. Near flakes have small $R_j$, so the $1/R_j^2$ gain (plus high snow reflectivity $\rho_s$) can make a clutter peak \textit{exceed} the true target peak $\rightarrow$ spurious foreground points, while the target only receives the un-occluded fraction $\theta_0/\theta$ (weakened/lost return).

\paragraph{Learned Domain Translation}
Data-driven domain translation: requires target data, can learn complex shifts but unrealiable, not very controllable nor interpretable. I.e., use a NN (image translator) to translate the source image into data representing the target domain.

\subparagraph{Paired Data / Supervised.} Works primarily when we have \textit{paired data}. Often misaligned pairs in reality (e.g. we can have driving data from the same location at different day times, yet the camera is not eactly the same, and dynamic objects have are different).

\subparagraph{Unpaired Data / Unsupervised.} Only two \textit{unaligned} sets $X_1$ (e.g.\ day), $X_2$ (e.g.\ night) --- \textit{no} correspondence $x_i\leftrightarrow y_i$. Realistic for driving (can't capture the exact same scene at two times of day).\\

\textbf{GANs.} \textbf{Generator} produces fake samples (from noise), \textbf{Discriminator} learns to tell real vs.\ fake $\rightarrow$ $G$ is pushed to match the target distribution.\\

\textbf{Conditional GAN for translation.} Feed a source image instead of noise: a \textbf{translation net} $F_{1\to2}$ maps $X_1\!\to\!X_2$, $D$ enforces outputs \textit{look like} domain~2. Only needs a domain-2 set, no pairs (e.g. translate dogs to cats).\\

\textbf{Mode Collapse.} GANs only force $F(x)\in X_2$, \textit{not} that the output \textbf{corresponds} to the input $\rightarrow$ $F$ can map all day images to one generic night image and still fool $D$ (scene content lost).\\

\textbf{CycleGAN.} Add a \textbf{cycle-consistency loss}: round-trip $x \xrightarrow{G} G(x) \xrightarrow{F} F(G(x))$ must reconstruct the input, $\lVert F(G(x))-x\rVert_1$ small. \textit{Intuition:} ties output to input \textbf{without} paired data $\rightarrow$ kills mode collapse (a collapsed mapping can't be inverted). E.g.\ \textbf{time-of-day translation} (day $\leftrightarrow$ twilight $\leftrightarrow$ night) to synthesize labeled adverse-condition data.

\paragraph{Hand-crafted Domain Transformation}
Transform in a fixed, \textit{non-learned} space (e.g.\ frequency): needs a target reference image; \textbf{controllable + interpretable}, but limited to the attributes of that one image and not always reliable.

\subparagraph{Fourier Domain Adaptation (FDA).} \textbf{Observation:} high-level object attributes (shape of cars, humans) barely change across domains (low-frequency shift). \textbf{Hypothesis:} aligning only the \textit{low-level statistics} of source to target is a good UDA baseline: FDA does this at the \textbf{spectral level/ frequency decomposition}, non-learned.\\

\textbf{Method.} Split the DFT into amplitude and phase, $\mathcal{F}(I)=\mathcal{F}^A(I)\,e^{j\mathcal{F}^P(I)}$. Low-level statistics live in the \textit{low-frequency amplitude} (near the center in fourier/ zero frequency), so \textbf{swap the source's low-freq amplitude for the target's}, keep the source \textit{phase} (= content). With low-pass mask $M_\beta(h,w)=\mathbf{1}_{(h,w)\in[-\beta H:\beta H,\,-\beta W:\beta W]}$ (replace the low-level frequencies):
\begin{equation*}
I_{s\to t}=\mathcal{F}^{-1}\!\Big(\big[\,M_\beta\,\mathcal{F}^A(I_t)+(1-M_\beta)\,\mathcal{F}^A(I_s)\,,\ \mathcal{F}^P(I_s)\big]\Big)
\end{equation*}
\textbf{Bandwidth $\beta$:} $\beta=0\Rightarrow$ output $=$ source; larger $\beta\Rightarrow$ closer to target \textit{but} more artifacts. \textit{Intuition:} cheap, interpretable "style transfer" $\rightarrow$ recolor the source to the target's look without touching structure. Bonus: varying $\beta$ yields \textbf{multiple diverse models} (one training set per $\beta$), which learned translators don't give for free.

Align the \textit{feature distributions of the source and the target domain}: impose constraints/restict feature maps.

\subsubsection{Output-Level Adaptation}
Align the \textit{outputs for the source and the target domain}: e.g. align entropies of the per pixel segmentation.
