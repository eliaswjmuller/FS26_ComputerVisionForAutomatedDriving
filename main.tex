\section{Part I: Fundamentals}
\subsection{Definitions}
\begin{definition}[Computer Vision]
Inverse algorithmic process to infer properties of the world based on sensory observations.
\end{definition}
\begin{definition}[Autonomous Car (ADS-DV)]
ADS-equipped vehicle designed for driverless operation within its given operational design domain.
\end{definition}
\begin{definition}[Automated Driving System (ADS)]
Hardware and software collectively capable of performing the entire DDT on a sustained basis, possibly limited to a specific ODD.
\end{definition}
\begin{definition}[Operational Design Domain (ODD)]
Conditions (environmental, geographical, temporal, traffic/roadway) under which a given ADS is designed to function.
\end{definition}
\begin{definition}[Dynamic Driving Task (DDT)]
All real-time operational and tactical functions for on-road driving (excludes strategic functions). Subtasks: (1) lateral control, (2) longitudinal control, (3) OEDR monitoring, (4) OEDR execution, (5) maneuver planning, (6) conspicuity enhancement.
\end{definition}
\begin{definition}[OEDR]
Object and Event Detection and Response. DDT subtasks 3--4: monitor driving environment via detection, recognition, classification, response preparation, and response execution.
\end{definition}
\begin{definition}[Driving Automation System]
Hardware and software capable of performing part or all of the DDT. $\text{ADS} \subseteq \text{Driving Automation Systems}$. Four core modules: (1) semantic and geometric \textbf{Perception} (2) \textbf{Prediction} from temporal perception (3) path and motion \textbf{Planning} via response determination (4) \textbf{Control}
\end{definition}

\subsection{Optical Autonomous Driving Sensors}
Countless sensor choices exist. We make the distinction of \textbf{external} (\textit{camera, lidar, radar, event camera, GPS, IMU, speedometer, ...}) and \textbf{internal} (\textit{camera, thermal camera, microphone}) sensors. Our focus further lies on the \textit{Optical Sensors}.\\

Optical Sensors can be categorized along the following features: 
\begin{itemize}
\item \textit{Wavelength}: spectral band of light the sensor operates in (e.g.\ visible, near-infrared, millimeter-wave).
\item \textit{Active or Passive}: whether the sensor emits its own signal (active) or relies on external illumination (passive).
\item \textit{Resolution}: spatial density of measurements; determines fineness of detail captured.
\item \textit{Latency / Frame Rate}: temporal frequency at which full measurements are acquired.
\item \textit{Field of View} (FOV): angular extent of the observable scene, horizontal and vertical.
\item \textit{Range}: maximum distance at which the sensor can reliably detect objects.
\end{itemize}

\subsubsection{Light Detection and Ranging (Lidar)}
\textbf{Characteristics}: Active sensor, near-infrared wavelength (905nm), direct range measurements via time of flight (TOF), i.e., timing of light emission until recapture of photons, large horizontal and fair vertical FOV (3D), fair resolution

\paragraph{Pulsed Lidar}
Multiple short laser pulses emitted in different directions and at different times. Each measurement: pulse emitter $\to$ beam steering $\to$ target $\to$ photodetector (APD). A timer measures the round-trip delay to infer range $R_0$

\subparagraph{Beam Configurations.}
Real beams diverge; receivers have finite FOV. Two configurations:
\begin{itemize}
    \item \textit{Coaxial}: transmitter and receiver share a common optical axis.
    \item \textit{Bistatic}: optical axes of transmitter and receiver are parallel but offset.
     The \textbf{crossover function} $\xi(R) \in [0,1]$ gives the fraction of the illuminated area observed by the receiver as a function of range. For bistatic: $\xi(R)=0$ at short range (no crossover), $\xi(R)=1$ beyond some $R_2$ (full crossover).
\end{itemize}

\subparagraph{Pulse Waveform (Sinusoidal Form).}
\begin{equation*}
    P_T(t) = \begin{cases} P_0 \sin^2\!\left(\dfrac{\pi}{2\tau_H}\, t\right), & 0 \leq t \leq 2\tau_H, \\ 0, & \text{otherwise.} \end{cases}
\end{equation*}
where $P_T(t)$: emitted radiant power, $P_0$: peak power, $\tau_H$: half-power pulse width ($10$--$20$\,ns), $c\tau_H$: spatial half-power width ($3$--$6$\,m).\\
Physically more realistic is the square-exponential form (left heavy, right tailed) -- omitted here. 

\subparagraph{Linear System Model.}
Received radiant power as a convolution of emitted power with the optical system impulse response:
\begin{equation*}
    P_R(R) = C_A \int_0^{2R/c} P_T(t)\, H\!\left(R - \frac{ct}{2}\right) dt
\end{equation*}
where $P_R(R)$: range-dependent received power, $H(R)$: impulse response of the optical system, $C_A$: system constant (independent of time and range). $ct/2$ maps time $t$ to the corresponding range (round-trip).
 
\textbf{Intuition:} $H(R)$ is what we would receive from a perfectly concentrated (Dirac) pulse; the actual response is a smeared version due to finite pulse width - ambiguity from temporal profile/ early or late photon.

\subparagraph{Impulse Response Decomposition.}
\begin{equation*}
    H(R) = H_C(R)\, H_T(R)
\end{equation*}
\begin{itemize}
    \item \textit{Optical channel}: $H_C(R) = \dfrac{T^2(R)}{R^2}\,\xi(R)$, \quad where $T(R) = \exp(-\alpha R)$ is the one-way transmission loss ($\alpha$ depends on visibility, assumes homogeneous scattering medium).
    \item \textit{Target}: $H_T(R) = \rho_0\,\delta(R - R_0)$ in clear weather, where $\rho_0$ is the differential reflectivity.
\end{itemize}
 
\subparagraph{Clear Weather Simplification.}
With $\alpha = 0$ (no attenuation -- clear air) and $\xi(R_0)=1$:
\begin{equation*}
    H(R) = \frac{1}{R_0^2}\,\rho_0\,\delta(R - R_0)
\end{equation*}
Substituting into the linear system model yields:
\begin{equation*}
    P_R(R) = \begin{cases} \dfrac{C_A P_0 \rho_0}{R_0^2}\, \sin^2\!\left(\dfrac{\pi(R-R_0)}{c\tau_H}\right), & R_0 \leq R \leq R_0 + c\tau_H, \\ 0, & \text{otherwise.} \end{cases}
\end{equation*}
 
\textbf{Intuition:} The target acts as a Dirac delta in range, so the convolution merely shifts and scales the emitted waveform — the received pulse preserves the shape of $P_T$, centered at $R_0$. The $1/R_0^2$ decay is the inverse-square law: emitted power spreads over a sphere of area $\propto R_0^2$. The proportionality to $\rho_0$ is why lidar returns per-point \textit{intensity} alongside range. The pulse occupies a range interval of width $c\tau_H$, so shorter pulses yield finer range resolution at the cost of less energy per pulse.

\subparagraph{Features of Pulsed Lidar.}
\begin{itemize}
    \item \textit{Ranging precision}: limited by temporal resolution of timing electronics. Typical $\sim 0.1$\,ns $\Rightarrow$ $\sim 1.5$\,cm.
    \item \textit{Maximum range}: limited by signal to noise ratio (SNR) (attenuation + detector noise/jitter): $100$--$200$\,m.
    \item \textit{Range--rate trade-off}: maximum unambiguous range is inversely proportional to pulse repetition rate (aliasing) -- upper limit of the integral
\end{itemize}

\subparagraph{Acquisition of 3D Point Clouds.}
Each pulse yields a single range measurement along a direction $(\alpha, \epsilon)$. To obtain a 3D point cloud, the sensor scans across a range of azimuth $\alpha$ and elevation $\epsilon$ values, which jointly define the FOV. Spherical-to-Cartesian conversion:
\begin{equation*}
    \begin{bmatrix} x \\ y \\ z \end{bmatrix}
    = \mathbf{h}^{-1}(r, \alpha, \epsilon)
    = \begin{bmatrix} r \cos\alpha \cos\epsilon \\ r \sin\alpha \cos\epsilon \\ r \sin\epsilon \end{bmatrix}
\end{equation*}
Each point additionally carries an \textit{intensity} value $= f(P_R(R_0))$, derived from the peak received power.
 
\subparagraph{Scanning Mechanisms.}
Taxonomy: non-scanning (flash) vs.\ scanning lidar. Scanning subdivides into non-mechanical (OPA, solid-state) and mechanical (MEMS, motorized optomechanical). Driving-relevant types are motorized mechanical and MEMS scanners.\\
 
\textit{Motorized mechanical}: nodding-mirror tilt controls vertical FOV; rotating housing ($>1000$ RPM) provides panoramic $360^\circ$ horizontal FOV. Modern systems use laser/photodiode arrays $\Rightarrow$ mechanical rotation only along azimuth. Horizontal angular resolution up to $0.1^\circ$, max range up to $200$\,m. \\
 
\textit{MEMS}: tiny oscillating mirrors (few mm diameter) steer individual pulses. 1D MEMS scans one angular dimension (needs laser array or motor for the other); 2D MEMS scans both. Resonant scanning: frequency $=$ resonant frequency of mirror, high frame rate, sensitive to environmental variation. Non-resonant: more robust, lower frame rate.
 
\begin{center}
\begin{tabular}{l|l}
    \textbf{Motorized Mechanical} & \textbf{MEMS} \\ \hline
    Panoramic $360^\circ$ horizontal FOV & Limited ($\sim 120^\circ$) horizontal FOV \\
    Uniform azimuthal scanning pattern & Complex / less interpretable pattern \\
    Bulky, vulnerable to wear & Small, no mechanical wear \\
    Imbalanced vert.\ vs.\ horiz.\ resolution & Balanced resolution \\
    Low frame rate ($10$--$20$\,Hz) & Slightly higher frame rate \\
    Expensive & Less expensive \\
\end{tabular}
\end{center}
 
\textbf{Measurement Errors in Lidar.}
\begin{itemize}
    \item \textit{Sensor}: timing jitter in arrival detection; rotary encoder uncertainty (angular error $\to$ distortion).
    \item \textit{Scene}: range ambiguity (multiple waveform maxima); non-Lambertian surfaces (absorptive $\to$ no return, specular $\to$ multi-path); motion distortion from finite frame rate (not rectifiable, unlike ego-motion).
    \item \textit{Optical medium}: attenuation (e.g.\ fog $\to$ reduced peak amplitude); backscattering (e.g.\ snow/rain $\to$ false peaks, spurious points).
    \item \textit{Light sources}: ambient NIR interference (minor); cross-talk from other lidars.
\end{itemize}

\subsubsection{Camera}
\textbf{Characteristics}: Passive sensor, visible spectrum (typically three sensitivity profiles / channels), very high resolution, fair frame rate, moderate FOV for perspective cameras, cheap.

\paragraph{Pinhole Camera Model.}
Ideal pinhole $=$ center of projection $\mathbf{O}_c$. All scene rays pass through a single point onto the image plane. Virtual image plane (in front of pinhole): image is not inverted. This yields a \textit{perspective projection}.

\paragraph{Homogeneous Coordinates.}
Unify representation of points and planes ($ax+by+cz-d=0$) by adding one dimension:
\begin{equation*}
    \mathbf{P} = \begin{pmatrix} x \\ y \\ z \\ 1 \end{pmatrix} \quad \text{(point)}, \qquad
    \boldsymbol{\Pi} = \begin{pmatrix} a \\ b \\ c \\ -d \end{pmatrix} \quad \text{(plane)},
\end{equation*}
where $\boldsymbol{\Pi} \cdot \mathbf{P} = 0$ iff $\mathbf{P}$ lies on $\boldsymbol{\Pi}$.


\paragraph{Geometric Transformations.}
Notation: ${}^{F}\mathbf{P}$ denotes coordinates of point $\mathbf{P}$ in frame $F$.
\begin{itemize}
    \item \textit{Translation}: ${}^{B}\mathbf{P} = {}^{A}\mathbf{P} + {}^{B}\mathbf{O}_A$, where ${}^{B}\mathbf{O}_A$ is the origin of frame $A$ expressed in frame $B$.
    \item \textit{Rotation}: ${}^{B}\mathbf{P} = {}^{B}_{A}\mathbf{R}\;{}^{A}\mathbf{P}$, where ${}^{B}_{A}\mathbf{R} = ({}^{B}\mathbf{i}_A \;\; {}^{B}\mathbf{j}_A \;\; {}^{B}\mathbf{k}_A)$ and $\mathbf{R}^T\mathbf{R} = \mathbf{I}$.
    \item \textit{Rigid / Euclidean}: ${}^{B}\mathbf{P} = {}^{B}_{A}\mathbf{R}\;{}^{A}\mathbf{P} + {}^{B}\mathbf{O}_A$. In homogeneous coordinates:
\end{itemize}
\begin{equation*}
    \begin{bmatrix} {}^{B}\mathbf{P} \\ 1 \end{bmatrix}
    = \begin{bmatrix} {}^{B}_{A}\mathbf{R} & {}^{B}\mathbf{O}_A \\ \mathbf{0}^T & 1 \end{bmatrix}
    \begin{bmatrix} {}^{A}\mathbf{P} \\ 1 \end{bmatrix}
\end{equation*}

\paragraph{3D to 2D Perspective Projection.}
Given a 3D point $\mathbf{P}_c = (x_c, y_c, z_c)^T$ in camera coordinates, projection to 2D homogeneous pixel coordinates $\tilde{\mathbf{x}}_s$:
\begin{equation*}
    \tilde{\mathbf{x}}_s \sim \mathbf{K}\, \mathbf{P}_c, \qquad
    \mathbf{K} = \begin{bmatrix} f & s & c_x \\ 0 & f & c_y \\ 0 & 0 & 1 \end{bmatrix}
\end{equation*}
where $f$: focal length, $(c_x, c_y)$: principal point, $K$ encoding the camera intinsics (5DOF)\\
 
\textbf{Intuition:} Derived from similar triangles starting with: $-(x_s - c_x)/x_c = f/(-z_c)$.\\
 
\textbf{Incorporating Extrinsics.} rigid transformation from world frame to camera frame: $ \mathbf{P}_c = [\mathbf{R} \mid \mathbf{t}]\, \tilde{\mathbf{P}}_w$

\paragraph{Complete Projection.} from 3D homogeneous world coordinates to 2D homogeneous image coordinates:
\begin{equation*}
    \tilde{\mathbf{x}}_s \sim \underbrace{\mathbf{K}\, [\mathbf{R} \mid \mathbf{t}]}_{\mathbf{C}}\; \tilde{\mathbf{P}}_w \qquad \text{ $\mathbf{C}$: $3 \times 4$ \textbf{camera matrix} (11 DOF: 5 intrinsics $+$ 6 extrinsics)
}
\end{equation*}

\paragraph{Photometric Image Formation.}
Digital imaging pipeline (scene radiance $\to$ optics $\to$ sensor $\to$ ISP $\to$ output), BRDF-based (Bidirectiional Reflectance Distribution Function) rendering equation, Lambertian vs.\ specular reflection. (details omitted)

\subsubsection{Radar}
\textbf{Characteristics}: Active sensor, $70$--$80$\,GHz (wavelengths $\sim$ few mm), measures \textit{range and velocity}. Robust to fog/rain/snow (wavelength $\gg$ particle size $\Rightarrow$ no scattering). Large horizontal FOV, very narrow vertical FOV. Low azimuthal resolution ($\approx$ measurements for $360\deg$), high range resolution. Low frame rate. Range: $200$--$600$\,m.

\paragraph{FMCW Scanning Radar.}
Two waveform types: fixed-frequency pulses and frequency-modulated continuous wave (FMCW). Output is discretized into $M$ azimuths $\times$ $N$ range bins; per-bin measurement is received power.\\
 
\textbf{Radar equation}: $P_{\text{received}} = \frac{P_{\text{transmitted}}\, G\, \sigma\, A_e}{16\pi^2\, r^4\, L}$
where $G$: transmitter gain, $\sigma$: radar cross section (RCS, depends on target area/orientation/material), $A_e$: effective antenna aperture, $L$: loss factor, $r$: range. Note $1/r^4$ decay (vs.\ $1/R_0^2$ for lidar) because radar uses separate opening angles for transmitter and receiver. This is the radar analogue of the pulsed lidar received-power model: it determines how SNR degrades with range and target properties, and thus the sensor's effective detection limits.\\

\textbf{FMCW Range and Velocity Measurement}:\\
\textit{Modulation pattern}: frequency of transmitted signal as a function of time.\\
\textit{Intermediate frequency (IF)}: difference between transmitted and received signal frequencies.\\
 
For a \textit{static target}, a single linear frequency sweep (chirp, i.e.\ frequency ramping at constant rate $df/dt$) yields range directly from the IF: $r = \frac{c\,\Delta f}{2\,(df/dt)}$ where $\Delta f$ is the IF.\\
 
For a \textit{moving target}, there is ambiguity between TOF shift and Doppler shift $\Delta f_v$. Resolved using repeating triangular modulation pattern (up-chirp IF $= \Delta f_1$, down-chirp IF $= \Delta f_2$):
\begin{equation}
    \Delta f_v = \frac{\Delta f_2 - \Delta f_1}{2}, \qquad
    \Delta f_r = \frac{\Delta f_1 + \Delta f_2}{2}
\end{equation}
$\Delta f_r$ gives range; $\Delta f_v$ gives radial velocity relative to sensor.\\

\textbf{Range-FFT}: FFT of the IF signal yields the power-range spectrum for each azimuth. Each frequency component in the IF corresponds to a specific range, so the FFT separates targets at different distances. Real targets are not point-like $\Rightarrow$ energy spreads across multiple range bins, unlike the ideal single-peak response.

\subsubsection{Sensor Distribution on Car}
Typical ADS platform uses $20$--$40$ sensors. Sensor placement affects covered FOV, cost, reliability, dirt management, serviceability, and crash survivability. Roof-mounted setups (common for robotaxis / data collection) are not acceptable for commercial vehicles. Changes in sensor 6D pose introduce dataset bias, potentially requiring full re-collection. Sensor suite composition is still under active definition in the industry.

\section{Part II: Semantic Perception}
\subsection{Semantic Segmentation}
\subsubsection{Definition and Basics}
\begin{definition}[Semantic Image Segmentation]
Given a predefined class set $\mathcal{C} = \{1, \dots, C\}$, assign every pixel $\mathbf{x}_i$ of an input image $\mathbf{I} \in \mathbb{R}^{H \times W \times 3}$ a single class label $y_i \in \mathcal{C}$. No differentiation between distinct instances of the same class (cf.\ instance/panoptic segmentation).
\end{definition}

\paragraph{Data Format.}
Supervised setup: dataset $\mathcal{D} = \{(\mathbf{I}^{(n)}, \mathbf{Y}^{(n)})\}_{n=1}^N$.
\begin{itemize}
    \item \textit{Input}: RGB image $\mathbf{I} \in \mathbb{R}^{H \times W \times 3}$.
    \item \textit{Output / ground truth}: one-hot tensor $\mathbf{Y} \in \{0,1\}^{H \times W \times C}$ with $\sum_{c=1}^{C} Y_{h,w,c} = 1$ for every pixel $(h,w)$. One channel per class.
\end{itemize}
Indexing convention: $h \in \{1,\dots,H\}$ row, $w \in \{1,\dots,W\}$ column, $c \in \{1,\dots,C\}$ class channel, $k$ used as a dummy class index in sums.

\paragraph{Prediction Format.}
Model outputs a logits tensor $\mathbf{Z} \in \mathbb{R}^{H \times W \times C}$. Per-pixel softmax gives the soft prediction tensor $\hat{\mathbf{Y}} \in [0,1]^{H \times W \times C}$:
\begin{equation*}
    \hat{Y}_{h,w,c} = \frac{\exp(Z_{h,w,c})}{\sum_{k=1}^{C} \exp(Z_{h,w,k})}, \qquad \hat{\mathbf{Y}}_{h,w,:} \in \Delta^{C-1}, \qquad \text{Hard prediction: $\hat{y}_{h,w} = \arg\max_c \hat{Y}_{h,w,c}$.}
\end{equation*}

\paragraph{Categorical Cross-Entropy Loss.}
Image-level loss as an explicit triple sum over spatial and class dimensions:
\begin{equation*}
    \mathcal{L}(\mathbf{I}) = -\frac{1}{HW} \sum_{h=1}^{H} \sum_{w=1}^{W} \sum_{c=1}^{C} Y_{h,w,c}\, \log \hat{Y}_{h,w,c}.
\end{equation*}
\textbf{Intuition:} One-hot $\mathbf{Y}$ collapses the inner $c$-sum to the single true-class term $-\log \hat{Y}_{h,w,\, y_{h,w}}$, leaving an average over $HW$ pixels.

\paragraph{Evaluation Metrics.}
\begin{itemize}
    \item \textit{Pixel accuracy}: fraction of correctly classified pixels. Skewed by frequent classes (sky, road).
    \item \textit{IoU (class $c$)}: $\mathrm{IoU}_c = \dfrac{|\hat{P}_c \cap P_c|}{|\hat{P}_c \cup P_c|}$, with $\hat{P}_c, P_c$ the predicted/GT pixel sets.
    \item \textit{mIoU}: $\frac{1}{C}\sum_c \mathrm{IoU}_c$. More stringent than pixel accuracy; mean over classes prevents head-class bias.
\end{itemize}

\subsubsection{Convolutional Neural Networks Primer}
\paragraph{Limitations of MLPs on Images.}
Vectorizing an $H \times W$ image as MLP input
\begin{enumerate}
    \item destroys 2D structure and local correlations; spatial invariances must be relearned;
    \item explodes the parameter count (e.g.\ $400\!\times\!250$ image, 10k hidden $\Rightarrow$ $\sim\!10^9$ parameters in the first layer alone).
\end{enumerate}

\paragraph{Discrete 2D Convolution.}
Slide the kernel $\mathbf{K} \in \mathbb{R}^{k_h \times k_w}$ over input $\mathbf{X}$; at each location multiply the overlapping pixel--weight pairs and sum:
\begin{equation*}
    (\mathbf{X} * \mathbf{K})[h,w] = \sum_{u=1}^{k_h} \sum_{v=1}^{k_w} X[h+u-1,\, w+v-1]\, K[u,v].
\end{equation*}
Formally a \textit{cross-correlation}; it differs from convolution proper by a kernel flip.

\textbf{Intuition:} The operation is \textit{local}, \textit{linear}, and \textit{shift-equivariant} (commutes with translation). The same parameters are used across all spatial locations -- \textit{parameter sharing}. The 2D structure of the input is preserved in the output.

\paragraph{Convolutional Layer: General Formulation.}
At layer $\ell$, the activation tensor of the previous layer
\begin{equation*}
    \mathbf{X}^{(\ell-1)} \in \mathbb{R}^{H_{\ell-1} \times W_{\ell-1} \times C_{\ell-1}}
\end{equation*}
is convolved with $C_\ell$ \textit{3D kernels}
\begin{equation*}
    \mathbf{K}^{(\ell, k)} \in \mathbb{R}^{k_H \times k_W \times C_{\ell-1}}, \qquad k = 1, \dots, C_\ell,
\end{equation*}
each paired with a scalar bias $b_k^{(\ell)}$. One distinct kernel channel exists for every input channel; the kernel is therefore a 3D tensor whose depth matches $C_{\ell-1}$.

\subparagraph{Single Convolutional Neuron (Linear Part).}
For output channel $k$ at spatial location $(h, w)$, the layer's linear response is a 2D convolution per input channel, summed across channels with the bias:
\begin{equation*}
    Z^{(\ell)}_{h, w, k} = b_k^{(\ell)} + \sum_{c=1}^{C_{\ell-1}} \sum_{u=1}^{k_H} \sum_{v=1}^{k_W} X^{(\ell-1)}_{h+u-1,\, w+v-1,\, c}\; K^{(\ell, k)}_{u, v, c}.
\end{equation*}
\textbf{Intuition:} The 2D cross-correlation generalizes to multi-channel inputs as a triple sum (two spatial $+$ one channel dim); the 3D kernel ``stacks'' the channel-wise correlations and the bias offsets the result. Parameters are shared across all spatial neurons of the same output channel.

\subparagraph{Element-wise Nonlinearity.}
A nonlinear activation $\sigma$ is applied independently to every scalar entry of $\mathbf{Z}^{(\ell)}$:
\begin{equation*}
    X^{(\ell)}_{h, w, k} = \sigma\!\left( Z^{(\ell)}_{h, w, k} \right).
\end{equation*}
The conv-and-bias step and the nonlinearity are conceptually separate sub-layers; $\sigma$ couples neither spatial positions nor channels.

\subparagraph{Output Tensor.}
Stacking the $C_\ell$ activated feature maps gives the 3D output tensor
\begin{equation*}
    \mathbf{X}^{(\ell)} \in \mathbb{R}^{H_\ell \times W_\ell \times C_\ell},
\end{equation*}
i.e.\ multiple features per spatial location, one per output channel. $C_\ell$ is a layer hyperparameter

\subparagraph{Parameter Count.}
Per output channel: $k_H k_W C_{\ell-1}$ kernel weights $+$ $1$ bias. Total parameters in layer $\ell$:
\begin{equation*}
    \#\theta^{(\ell)} = C_\ell \,(\, k_H\, k_W\, C_{\ell-1} + 1\,).
\end{equation*}
\textbf{Intuition:} Cost scales with kernel footprint and the product of input/output channels, \textit{not} with the spatial size of the input -- this is what makes convolutional layers tractable on high-resolution images.

\paragraph{Pooling Layer.}
Non-parametric, (sub)differentiable downsampling. Hyperparameters: window size, stride. Max pooling provides robustness to local input variations and enables higher feature dimensionality without spatial-memory explosion.

\paragraph{Practical Variations.}
\begin{itemize}
    \item \textit{Padding}: valid convolution shrinks spatial dims. \textit{Same} mode pads by $\lfloor k/2 \rfloor$ (odd $k$) to preserve them. Modes: zero, replicate, reflect.
    \item \textit{Strided convolution}: kernel slides with step $s>1$. Motivation: faster receptive-field growth, spatial downsampling integrated into the conv. Often in early layers but pooling more common for downsampling. Higher stride $\rightarrow$ faster increase of receptive field size
\end{itemize}
Output spatial size (per dim):
\begin{equation*}
    H_{\mathrm{out}} = \left\lfloor \frac{H_{\mathrm{in}} + 2p - k}{s} \right\rfloor + 1,
\end{equation*}
where $H_{\mathrm{in}}$: input size, $k$: kernel size, $p$: padding width per border, $s$: stride.

\paragraph{Receptive Field.}
Region of the input on which a neuron's output depends. Grows with depth, kernel size, and stride. Deep CNN layers eventually attain global receptive fields covering the entire image.

\paragraph{CNN for Image Classification.}
Architecture: (conv $+$ activation $+$ pool)$^L \to$ flatten $\to$ FC layers $\to$ softmax. Final FC layers learn global interdependencies needed for the image-level prediction.

\subsubsection{Encoder--Decoder Paradigm}
\paragraph{Naive: Sliding-Window Classification.}
Extract a patch around each pixel, classify its center with a CNN. \\
\textit{Problems}: massive redundant computation; no global context (esp.\ for large objects). Does not scale to HR images.

\paragraph{End-to-End Architectures.}
Single forward pass from $\mathbf{I}$ to full label map $\hat{\mathbf{Y}}$. Goal: share computation across all spatial locations. Strategy: adapt classification CNNs to dense prediction.

\paragraph{Three Challenges.} (I) \textit{Reduced feature resolution} from repeated striding/pooling, (II) \textit{Multi-scale objects}, (III) \textit{Reduced localization accuracy} from CNN invariance

\paragraph{Fully Convolutional Networks (FCN).}
Replace the FC layers of a classification network with convolutional layers $\Rightarrow$ coarse high-level feature maps at low resolution. Per-pixel prediction at original image resolution then requires upsampling.

\paragraph{Naive Resolution Recovery.}
Remove pooling/striding entirely. Requires larger kernels for the same receptive field; every feature map $\mathbf{X}^{(\ell)}$ stays at $HW$ instead of shrinking geometrically, so activation memory $\sum_\ell H_\ell W_\ell C_\ell = HW \sum_\ell C_\ell$ scales linearly with depth (no geometric decay), and the same factor inflates per-layer FLOPs $\Rightarrow$ heavy compute/memory cost and overfitting risk.

\paragraph{Encoder--Decoder.}
\begin{itemize}
    \item \textit{Encoder}: maps $\mathbf{I}$ to a low-resolution, high-dimensional feature tensor; aggregates context.
    \item \textit{Decoder}: upsamples encoder features back to image resolution while reducing feature dimensionality.
\end{itemize}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.85\linewidth]{img/encoder_decoder.png}
    \caption{Encoder--decoder schematic. Spatial resolution contracts inside the encoder (high$\to$low) and is recovered by the decoder (low$\to$high); feature dimensionality follows the opposite trajectory.}
    \label{fig:encoder_decoder}
\end{figure}

\paragraph{In-Network Upsampling.}
\subparagraph{Unpooling.} During encoder max-pooling, store argmax positions in \textit{switch variables}; in the decoder, place each activation back at its recorded location. Produces an enlarged but \textit{sparse} feature map. Not learnable + symetric architecture for corresponding pairs.

\subparagraph{Transpose Convolution (Deconvolution).} Learnable upsampling for dense feature maps. Each input scalar weights a copy of the kernel; the weighted kernel is placed into the output at a location determined by the input position and the stride. Where output positions overlap across different input pixels, the contributions are summed. (Crop might be required to get exact target size)

    \includegraphics[width=0.65\linewidth]{img/transpose_conv_1d.png}
    \caption{1D transpose convolution. Input $(a,b)$ weights copies of filter $(x,y,z)$; the weighted filters are placed into the output and summed where they overlap (here only the middle entry, $az + bx$).}
    \label{fig:transpose_conv_1d}
\end{figure}

\subsubsection{Fundamental Architectures and Modules}
\paragraph{U-Net.}
Symmetric encoder--decoder with \textit{skip connections}: at each decoder stage, high-resolution encoder features are concatenated with upsampled decoder features along the channel dim, followed by convolutions.
\begin{itemize}
    \item Contracting path: $\downarrow$ ``Where'', $\uparrow$ ``What'' (context).
    \item Expansive path: transpose convolutions $+$ concatenations recover spatial detail.
\end{itemize}
\textbf{Intuition:} Skip connections inject fine-grained localization into the decoder, addressing the resolution challenge.
\begin{figure}[H]
    \centering
    \includegraphics[width=0.95\linewidth]{img/unet.png}
    \caption{U-Net: U-shaped encoder--decoder with skip connections (yellow arrows) copying high-resolution features from the contracting to the expansive path.}
    \label{fig:unet}
\end{figure}

\paragraph{Atrous (Dilated) Convolution.}
The \textit{effective} receptive field is smaller than the formal one: border connections are sparse. Atrous convolution inserts holes between kernel taps. For rate $r$:
\begin{equation*}
    (\mathbf{X} *_r \mathbf{K})[i,j] = \sum_{u,v} X[i+ru,\, j+rv]\, K[u,v].
\end{equation*}
Standard convolution corresponds to $r=1$. Stacking atrous convolutions with growing $r$ yields \textit{exponential} expansion of receptive field size without resolution loss or coverage gaps. \\

\textbf{Intuition:} Less need for pooling: receptive fields grow without spatial downsampling, preserving feature map resolution for dense prediction.
\begin{figure}[H]
    \centering
    \includegraphics[width=0.85\linewidth]{img/atrous_rates.png}
    \caption{Atrous convolution at rates $r=1,2,4$ (red dots: kernel taps). Sampling spacing scales with $r$, expanding the receptive field exponentially while the number of weights stays fixed.}
    \label{fig:atrous_rates}
\end{figure}

\paragraph{Attention.}
Soft, content-based indexing across all tokens (DB analogy). For tokens $\mathbf{X} \in \mathbb{R}^{N \times d}$, learned projections $\mathbf{W}_Q, \mathbf{W}_K, \mathbf{W}_V$ produce
\begin{equation*}
    \mathbf{Q} = \mathbf{X}\mathbf{W}_Q, \quad \mathbf{K} = \mathbf{X}\mathbf{W}_K, \quad \mathbf{V} = \mathbf{X}\mathbf{W}_V.
\end{equation*}
\begin{itemize}
    \item \textit{Query}: representation of the token used to compare it with other tokens.
    \item \textit{Key}: reference representation by which other tokens are compared against the token.
    \item \textit{Value}: output representation contributed by the token when it gets ``indexed''.
\end{itemize}
Scaled dot-product attention:
\begin{equation*}
    \mathrm{Attn}(\mathbf{Q}, \mathbf{K}, \mathbf{V}) = \mathrm{softmax}\!\left(\frac{\mathbf{Q}\mathbf{K}^\top}{\sqrt{d_k}}\right) \mathbf{V}.
\end{equation*}
Softmax taken along rows (per-query distribution). Division by $\sqrt{d_k}$ keeps gradient magnitudes well-scaled. Complexity: $\mathcal{O}(N^2 d)$ -- quadratic in tokens.\\

\textbf{Intuition:} Query/key match measures relevance of every other token; the row-softmax weights aggregate values $\Rightarrow$ each token is a learned linear mixture of all others. Globally connected receptive field at every layer.

\paragraph{Transformer Block.} Multi-head self-attention (concatenate $h$ parallel attention heads $\to$ linear projection) followed by a position-wise MLP. Residual connections and layer normalization around each sub-layer.

\paragraph{Transformers for Dense Prediction.}
ViT-style large patches lose spatial precision $\Rightarrow$ poor for segmentation. Naive fix: restrict attention to local windows of $M \times M$ patches. Complexity becomes \textit{linear} in the number of patches (for fixed $M$). Drawback: receptive field bounded by the window across all layers.

\paragraph{Swin Transformer.}
\textit{Shifted-window} attention: alternate between W-MSA (window-based multi-head self-attention) and SW-MSA (shifted-window MSA). The window grid in SW-MSA is offset relative to W-MSA, so a query $q$'s receptive field at layer $\ell+1$ is the union of layer-$\ell$ windows that overlap with $q$'s current local window.\\

\textbf{Intuition:} With enough shifts, local windowed attention reaches large receptive fields rapidly while keeping per-layer cost linear.

\paragraph{Hierarchical Structure.}
\textit{Patch merging} concatenates features of each $2\!\times\!2$ neighboring patch group and applies a linear projection: tokens reduce by $4\times$ ($2\times$ spatial downsampling), output dim set to $2C$. Yields a multi-scale feature pyramid; Swin can serve as a drop-in CNN-backbone replacement for the encoder of standard segmentation networks.
